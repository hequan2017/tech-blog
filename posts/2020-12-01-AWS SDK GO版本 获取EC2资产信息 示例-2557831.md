> **内容介绍**
>
> 用 AWS SDK for Go 查询指定 Region 下的 EC2 实例资产信息：通过静态 AK/SK 创建会话，调用 `DescribeInstances` 遍历 Reservations 与 Instances，打印实例 ID、Name 标签、内网 IP 等关键字段，适合采集云主机清单导入 CMDB。

> **技术备注**
>
> 示例基于 AWS SDK for Go v1（2020 年）；AWS 已推出 SDK for Go v2（`aws-sdk-go-v2`），模块化、支持 IAM 角色链与新版 API，建议新项目使用 v2。原文 import 路径被截断，实际应为 `github.com/aws/aws-sdk-go/...`。硬编码凭证仅适合测试，生产环境请改用 IAM Role 或环境变量。

---

```go
package main

import (
	"fmt"

	"github.com/aws/aws-sdk-go/aws"
	"github.com/aws/aws-sdk-go/aws/credentials"
	"github.com/aws/aws-sdk-go/aws/session"
	"github.com/aws/aws-sdk-go/service/ec2"
)

func GetInstances() (*ec2.DescribeInstancesOutput, error) {
	sess, _ := session.NewSession(&aws.Config{
		Credentials: credentials.NewStaticCredentials("111111111111", "2222222222222222222222222222222", ""),
		Region:      aws.String("ap-east-1"),
	})
	svc := ec2.New(sess)
	result, err := svc.DescribeInstances(nil)

	if err != nil {
		return nil, err
	}
	return result, nil
}

func main() {

	result, err := GetInstances()
	if err != nil {
		fmt.Println("Got an error retrieving information about your Amazon EC2 instances:")
		fmt.Println(err)
		return
	}
	for _, r := range result.Reservations {
		//fmt.Println(r) 打印所有返回信息
		for _, i := range r.Instances {
			fmt.Println("   " + *i.InstanceId + "   " +  *i.Tags[0].Value+"   " + *i.PrivateIpAddress + "   " + * + "   " + *i.InstanceType)
		}
	}
}
```

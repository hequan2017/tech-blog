> **内容介绍**
>
> 本文是一个 Go 小工具的完整源码:用 xorm 连接 MySQL,查询 token 表中
> type=0 记录的 up_time 字段,与当前时间(Asia/Shanghai 时区)求差;
> 若 token 更新时间落后系统时间超过 20 分钟,即通过钉钉机器人 webhook
> (http.Post 发 JSON)推送报警;另用 defer + recover 捕获 panic,把
> 程序报错也发到钉钉。

> **技术备注**
>
> 依赖为 go-xorm/xorm 与 go-sql-driver/mysql。原文两处 import 路径的
> `github.com/` 前缀、以及 `ntm := tm.In(loc)` 中的 `tm.In` 曾被编辑器
> 吃掉,已补全。`ioutil.ReadAll` 自 Go 1.16 起弃用,新代码请改用
> `io.ReadAll`;另注意时间布局 `03:04:05` 是 12 小时制,数据库存的是
> 24 小时制时间时应写 `15:04:05`,否则小时超过 12 会解析失败。

## 1. 监控数据库字段并钉钉报警

程序逻辑:连接数据库 → 查询 token 表的 up_time → 与系统当前时间求差 → 相差超过 20 分钟就调用钉钉 webhook 报警。

```go
package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	_ "github.com/go-sql-driver/mysql"
	"github.com/go-xorm/xorm"
	"io/ioutil"
	"net/http"
	"time"
)

var engine *xorm.Engine

const webhook_url = "https://oapi.dingtalk.com/robot/send?access_token=xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx"

func dingtoinfo(s string) bool {
	content, data := make(map[string]string), make(map[string]interface{})
	content["content"] = s
	data["msgtype"] = "text"
	data["text"] = content
	b, _ := json.Marshal(data)

	resp, err := http.Post(webhook_url,
		"application/json",
		bytes.NewBuffer(b))
	if err != nil {
		fmt.Println(err)
	}
	defer resp.Body.Close()
	body, _ := ioutil.ReadAll(resp.Body)
	fmt.Println(string(body))
	return true
}

func recoverName() {
	if r := recover(); r != nil {
		res := fmt.Sprintf("%s %s", "监控程序 报错:", r)
		dingtoinfo(res)
	}
}

func main() {
	defer recoverName()
	var err error
	engine, err = xorm.NewEngine("mysql", "root:xxxxxxxxxxxxxxx@tcp(127.0.0.1:3306)/new?charset=utf8")

	if err != nil {
		panic(err.Error())
	}
	gsql := " select up_time from token where type = 0 ;"
	gres, gerr := engine.Query(gsql)

	if gerr != nil {
		panic(gerr.Error())
	}
	var data string
	for _, v := range gres {
		data = string(v["up_time"])
	}
	loc, _ := time.LoadLocation("Asia/Shanghai")
	nowTime := time.Now().In(loc)

	tm, _ := time.Parse("2006-01-02 03:04:05", data)
	ntm := tm.In(loc)
	subM := nowTime.Sub(ntm)
	fmt.Println("token更新时间 与 系统时间 相差", subM)
	subtract := int(subM.Minutes())
	if subtract > 20 {
		dingtoinfo("异常, 请及时登录查看 !!!")
	}
}
```

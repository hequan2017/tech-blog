## 1. 基本权限与 chmod / chown

r、w、x 分别对应数字 4、2、1：

```shell
[root@localhost ~]# chmod 575 a.txt
[root@localhost ~]# ls -l a.txt
-r-xrwxr-x 1 root root 0 3月  27 19:09 a.txt
```

递归与字符模式：

```shell
# chmod -R 777 xx/                  # 递归更改权限
# chmod u=rwx,g=rwx,o=rwx xx        # a=rwx 表示所有用户，还可用 + -
# chmod a-w xx
# chown root:root xx
```

cut 按列截取（-d 分隔符、-f 数字选列、-c/-b 按字符）：

```shell
# who | cut -b 10-14
tty1
pts/0
# who | cut -b 3-
# who | cut -c 3-5
# cat /etc/passwd | head -n 5 | cut -d : -f 1
root
bin
daemon
adm
lp
```

指定属组创建用户（chgrp 同理）：

```shell
# useradd yy -g he
$ ll
-rw-r--r-- 1 xx he 15 3月  27 20:20 a
```

## 2. umask 默认权限

删除文件的权限受父目录权限控制，和文件本身权限无关。

```shell
# umask
0022
```

- 文件：666-022
- 目录：777-022

普通用户的 umask 未必是 002，除非满足下面的条件：

```shell
# vi /etc/profile
if [ $UID -gt 199 ] && [ "`id -gn`" = "`id -un`" ]; then
    umask 002
else
    umask 022
fi
```

```shell
# vi /etc/login.defs
UMASK           077     # 针对家目录
```

## 3. 特殊权限：setuid / setgid / 粘滞位

- setuid：让普通用户以 root 的角色运行（典型例子 passwd）
- setgid：目录下面的文件自动继承目录的属组
- 粘滞位（sbit）：共享目录里只能属主和 root 才能删除，其他人动不了别人的东西

给无执行位的文件加 suid 会显示大写 S：

```shell
# chmod u+s /etc/passwd
-rwSr--r--
```

查找系统上的 setuid 程序：

```shell
# find /usr/bin -type f -perm 4755 | xargs ls -l
-rwsr-xr-x. 1 root root 54240 1月  30 2012 /usr/bin/at
-rwsr-xr-x. 1 root root 66352 12月  8 2011 /usr/bin/chage
-rwsr-xr-x. 1 root root 51784 11月 23 2013 /usr/bin/crontab
-rwsr-xr-x. 1 root root 71480 12月  8 2011 /usr/bin/gpasswd
-rwsr-xr-x. 1 root root 36144 12月  8 2011 /usr/bin/newgrp
-rwsr-xr-x. 1 root root 30768 2月  22 2012 /usr/bin/passwd
-rwsr-xr-x  1 root root 27576 9月  20 2013 /usr/bin/pkexec
```

setgid 用于组共享目录：

```shell
# chown .xx /home/admins/
# chmod 770 /home/admins/
# chmod 2770 /home/admins/      # 目录下面的文件自动继承目录的属组
# ls -ld /home/admins/
drwxrws--- 2 root xx 4096 3月  27 22:07 /home/admins/
# ls -l /home/admins/xxxxx
-rw--w--w- 1 root xx 0 3月  27 22:09 /home/admins/xxxxx
```

数字表示：setuid 4000 + setgid 2000 = 6755：

```text
-rwsr-xr-x                -rwxr-sr-x
```

```shell
# chmod 4755 a                  # 结果 -rwsr-xr-x
# chmod u+s a                   # chmod g+s a
```

与 x 重合的时候就是小写的 s。

粘滞位为 1000，典型如 /tmp：

```shell
# ls -ld /tmp/
drwxrwxrwt. 3 root root 4096 3月  27 21:46 /tmp/
# chmod 7755 test
```

package jielv

import (
  "encoding/xml"
  "/PuerkitoBio/goquery"
  "io/ioutil"
  "log"
  "net/http"
  "strings"
)

func getXML() []byte {
  url := "http://"
  resp, _ := http.Get(url)
  defer resp.Body.Close()
  data, _ := ioutil.ReadAll(resp.Body)
  return data
}

type  Test struct {
  
}

func main() {

  re := getXML()
  dom, err := goquery.NewDocumentFromReader(strings.NewReader(string(re)))
  if err != nil {
    log.Fatalln(err)
  }
  var s string
  dom.Find("pre").Each(func(i int, selection *goquery.Selection) {
    s = s + selection.Text()
  })
  var city Test
  xml.Unmarshal([]byte(s), &city)
}

> **内容介绍**
>
> 本文给出拨号管理系统 go-ppp 项目中 ppp 包的完整代码：定义
> PppData（账号、密码、带宽、Vlanid、数量、Mac、网卡）与分页
> 查询参数等结构体，基于 GORM 实现拨号账号的增删改查：
> Create、Delete、按 ids 批量删除、Update、按 id 单查，以及
> GetPppInfoList 分页列表，支持按账号关键字模糊搜索。

> **技术备注**
>
> 写操作由 db.Create / Delete / Save 一行完成，批量删除用
> "id in ?" 条件；列表查询以 Model 起链，先 Count 取总数，再
> Limit/Offset 分页、Find 取结果。方法签名沿用 (err, data)
> 的旧式返回顺序，PageInfo / IdsReq / GetXxxInfoList 这套结构
> 与 gin-vue-admin 自动生成的 service 模板一致。注意：列表按
> info.Account 传关键字，SQL 中匹配的列名是 sn；PppData 的
> Vlanid / Mac / Nic 等字段，对应本系列《创建ppp拨号》脚本里
> 建 VLAN、macvlan 及 ppp 配置所需的入参。

## 1. ppp 包完整代码

整个文件即 ppp 包：先定义数据结构与分页参数，再给出六个 CRUD 方法：

```go
package ppp

import (
    "gorm.io/gorm"
)

type PppService struct {
}

type PppSearch struct {
    PppData
    PageInfo
}

type IdsReq struct {
    Ids []int `json:"ids" form:"ids"`
}

type PageInfo struct {
    Page     int    `json:"page" form:"page"`         // 页码
    PageSize int    `json:"pageSize" form:"pageSize"` // 每页大小
    Keyword  string `json:"keyword" form:"keyword"`   //关键字
}

type PppData struct {
    Account   string
    Password  string
    Bandwidth string
    Vlanid    int
    Count     int
    Mac       string
    Nic       string
}

var db *gorm.DB

func (PppService *PppService) CreatePpp(ppp PppData) (err error) {
    err = db.Create(&ppp).Error
    return err
}

func (PppService *PppService) DeletePpp(ppp PppData) (err error) {
    err = db.Delete(&ppp).Error
    return err
}

func (PppService *PppService) DeletePppByIds(ids IdsReq) (err error) {
    err = db.Delete(&[]PppData{}, "id in ?", ids.Ids).Error
    return err
}

func (PppService *PppService) UpdatePpp(ppp PppData) (err error) {
    err = db.Save(&ppp).Error
    return err
}

func (PppService *PppService) GetPpp(id uint) (err error, ppp PppData) {
    err = db.Where("id = ?", id).First(&ppp).Error
    return
}

func (PppService *PppService) GetPppInfoList(info PppSearch) (err error, list interface{}, total int64) {
    limit := info.PageSize
    offset := info.PageSize * (info.Page - 1)
    // 创建db
    db1 := db.Model(&PppData{})
    var ppps []PppData
    // 如果有条件搜索 下方会自动创建搜索语句
    if info.Account != "" {
        db1 = db1.Where("sn LIKE ?", "%"+info.Account+"%")
    }
    err = db1.Count(&total).Error
    if err != nil {
        return
    }
    err = db1.Limit(limit).Offset(offset).Find(&ppps).Error
    return err, ppps, total
}
```

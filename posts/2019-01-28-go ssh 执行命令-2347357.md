> **内容介绍**
>
> 用 Go 的 golang.org/x/crypto/ssh 包封装一个简单的 SSH 客户端：通过密码认证建立连接，执行远程命令并用 `CombinedOutput` 同时获取标准输出和标准错误。适合批量运维小工具的基础骨架。
>
> **技术备注**
>
> 生产环境不建议用 `HostKeyCallback` 返回 nil 的方式跳过主机密钥校验（存在中间人攻击风险），x/crypto/ssh 新版提供 `gossh.InsecureIgnoreHostKey()` 仅供测试；正式使用应配合 knownhosts 校验，敏感环境建议改用密钥认证。

### go ssh  执行命令

```go
package main

import (
   "fmt"
   "net"

   gossh "golang.org/x/crypto/ssh"
)

type Cli struct {
   user    string
   pwd     string
   addr    string
   client  *gossh.Client
   session *gossh.Session
   LastResult string
}

func (c *Cli) Connect() (*Cli, error) {
   config := &gossh.ClientConfig{}
   config.SetDefaults()
   config.User = c.user
   config.Auth = []gossh.AuthMethod{gossh.Password(c.pwd)}
   config.HostKeyCallback = func(hostname string, remote net.Addr, key gossh.PublicKey) error { return nil }
   client, err := gossh.Dial("tcp", c.addr, config)
   if nil != err {
      return c, err
   }
   c.client = client
   return c, nil
}

func (c Cli) Run(shell string) (string, error) {
   if c.client == nil {
      if _,err := c.Connect(); err != nil {
         return "", err
      }
   }
   session, err := c.client.NewSession()
   if err != nil {
      return "", err
   }
   defer session.Close()
   buf, err := session.CombinedOutput(shell)

   c.LastResult = string(buf)
   return c.LastResult, err
}

func main() {
   cli := Cli{
      user:"root",
      pwd: "xxxxxxxxxxxxxxxxxxxxx",
      addr: "192.168.100.50:22",
   }
   output, err := cli.Run("pwd")
   fmt.Printf("%v\n%v", output, err)
}
```

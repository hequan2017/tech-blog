```plain
# 下载docker-compose文件
git clone https:///zabbix/zabbix-docker.git 

# 启动服务
cd zabbix-docker
git checkout 6.0  # 切换到6.0 版本，进行6.0版本安装
ln -s docker-compose_v3_alpine_mysql_latest.yaml docker-compose.yaml # 创建软链接
docker-compose up -d
```

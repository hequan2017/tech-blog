```g

package main

import (
	"fmt"
	"log"
	"time"

	"/shirou/gopsutil/cpu"
)

func main() {
	// 获取CPU核心数
	cpuCount, err := cpu.Counts(true)
	if err != nil {
		log.Fatal("Failed to get CPU core count:", err)
	}

	// 设置采集的间隔时间
	interval := time.Second

	for {
		// 采集每个核心的CPU使用率
		cpuPercent, err := cpu.Percent(interval, true)
		if err != nil {
			log.Println("Failed to get CPU usage:", err)
			continue
		}

		// 显示每个核心的使用率
		for i := 0; i < cpuCount; i++ {
			fmt.Printf("CPU Core %d Usage: %.2f%%\n", i, cpuPercent[i])
		}

		fmt.Println("------------------")

		// 等待一段时间
		time.Sleep(interval)
	}
}
```

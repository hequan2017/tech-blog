#### api

```pythonimport request from '@/utils/request'

export function login(data) {
  return request({
    url: '/auth',
    method: 'post',
    data
  })
}

export function getInfo(token) {
  return request({
    url: '/api/v1/userInfo',
    method: 'get',
    params: { token }
  })
}
```

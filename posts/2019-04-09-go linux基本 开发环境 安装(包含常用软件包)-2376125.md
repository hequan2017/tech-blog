```
yum install go -y

mkdir -p /data/go
echo export GO111MODULE=on >> /etc/profile
echo export GOPATH=/data/go >> /etc/profile
echo export GOPROXY=https://goproxy.io >> /etc/profile
source /etc/profile
```

```
git clone https:///golang/sys.git $GOPATH/src//golang/sys
git clone https:///golang/net.git $GOPATH/src//golang/net
git clone https:///golang/text.git $GOPATH/src//golang/text
git clone https:///golang/lint.git $GOPATH/src//golang/lint
git clone https:///golang/tools.git $GOPATH/src//golang/tools
git clone https:///golang/crypto.git $GOPATH/src//golang/crypto
git clone https:///golang/sync.git $GOPATH/src//golang/sync
git clone https:///golang/time.git $GOPATH/src//golang/time
git clone https:///golang/image.git $GOPATH/src//golang/image

mkdir -p  $GOPATH/src/
ln -s  $GOPATH/src//golang/  $GOPATH/src//x 
```

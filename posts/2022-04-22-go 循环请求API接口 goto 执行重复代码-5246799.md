```
package main

import "fmt"

func main() {
  repeat := 0
  pagesize := 10
  count := 150

  /* 循环 */
LOOP:
  firstResult := repeat * pagesize
  for repeat < 1000 {
    fmt.Println("一共150个数据，一次请求10个， 本次是第", repeat, "请求", " firstResult 是: ", firstResult)
    // 这里可以去获取 count的 数据量
    if count > pagesize {
      count = count - pagesize
      repeat = repeat + 1
      goto LOOP
    } else {
      break
    }
  }
}
```

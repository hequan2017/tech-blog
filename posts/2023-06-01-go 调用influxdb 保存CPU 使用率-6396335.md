> **内容介绍**
>
> 本文用 gopsutil 逐核采集 CPU 使用率（采样间隔 1 秒），
> 经 influxdb-client-go/v2 把每个核心的使用率组织成
> cpu 数据点逐条写入 InfluxDB，文末附完整示例代码。

> **技术备注**
>
> influxdb-client-go/v2 采用 org/bucket 模型：
> WriteAPIBlocking("test", "test") 两个参数依次是组织
> 与桶，区别于 v1 SDK 的 database/retention policy 写法；
> 数据点由 SDK 转成 Line Protocol 写入。
> cpu.Percent(interval, true) 会先阻塞一个采样间隔，
> 再返回各核心使用率；cpu.Counts(true) 为逻辑核心数。
> NewClient("http://", "1") 为脱敏占位，实际应替换为
> 服务地址与 API token；原文 "id": id 引用了未定义
> 变量，已按参数改为 machineid，并补回 import 被吃掉
> 的 github.com/ 前缀。

---

## 1. 完整示例代码

```go
package main

import (
    "context"
    "fmt"
    "github.com/influxdata/influxdb-client-go/v2/api/write"
    "github.com/shirou/gopsutil/cpu"
    "log"
    "time"

    influxdb2 "github.com/influxdata/influxdb-client-go/v2"
)

func main(machineid string) {
    // 连接到 InfluxDB
    client := influxdb2.NewClient("http://", "1")

    // 创建写入 API
    writeAPI := client.WriteAPIBlocking("test", "test")

    cpuCount, err := cpu.Counts(true)
    if err != nil {
        log.Fatal("Failed to get CPU core count:", err)
    }
    // 设置采集的间隔时间
    interval := time.Second

    // 采集每个核心的CPU使用率
    cpuPercent, err := cpu.Percent(interval, true)
    if err != nil {
        log.Println("Failed to get CPU usage:", err)
    }

    // 创建数据点数组
    points := make([]*write.Point, cpuCount)
    for i := 0; i < cpuCount; i++ {
        fmt.Println(i, cpuPercent[i])
        // 创建数据点
        points[i] = influxdb2.NewPoint("cpu",
            map[string]string{"tag1": "id", "tag2": "cpu", "tag3": "percent"},
            map[string]interface{}{"id": machineid, "cpu": i, "percent": cpuPercent[i]},
            time.Now())
        // 批量写入数据点
        err = writeAPI.WritePoint(context.Background(), points[i])
        if err != nil {
            log.Println("Failed to write data points:", err)
        }
    }

    // 关闭连接
    client.Close()
}
```

> 先保存当前一分钟的流量数据，然后用后面一分钟的数据 相减，就是当前一分钟的网速。

IOCountersStat, _ := gopsutil.IOCounters(true)
    loc, _ := time.LoadLocation("Asia/Shanghai")
    now := time.Now()
    NewNow := now.In(loc).Format("2006-01-02 15:04:00")
    NewDt, _ := time.ParseInLocation("2006-01-02 15:04:05", NewNow, loc)
    NewDtNumber := NewDt.Unix()

    var data []Net
    for _, v := range IOCountersStat {
        if v.Name == "lo" || v.Name == "docker0" {
            continue
        }
        var d Net
        d.Time = uint64(NewDtNumber)
        d.Hostname = name
        d.Machineid = machineid
        d.BytesSent = v.BytesSent
        d.BytesRecv = v.BytesRecv
        d.PName = v.Name
        d.IsUpdate = false
        data = append(data, d)
    }

timeList := []int64{60, 120, 180, 240, 300}

    for _, t := range timeList {
        var readPppData []Net
        _ = db.Find(&readPppData, "time = ?", NewDtNumber-t)
        if len(readPppData) == 0 {
            continue
        } else {
            for _, v := range readPppData {
                if v.Time == uint64(NewDtNumber-t) {
                    for i, v1 := range data {
                        if v.PName == v1.PName {
                            if data[i].BytesSent > v.BytesSent {
                                data[i].Outbytes = data[i].BytesSent - v.BytesSent
                                out, _ := strconv.ParseFloat(fmt.Sprintf("%.2f", float64(data[i].BytesSent-v.BytesSent)/float64(t)/1000/1000*8), 64)
                                data[i].Outflow = out
                            }
                            if data[i].BytesRecv > v.BytesRecv {
                                data[i].Inbytes = data[i].BytesRecv - v.BytesRecv
                                in, _ := strconv.ParseFloat(fmt.Sprintf("%.2f", float64(data[i].BytesRecv-v.BytesRecv)/float64(t)/1000/1000*8), 64)
                                data[i].Inflow = in
                            }
                        }
                    }
                }
            }
            return data
        }

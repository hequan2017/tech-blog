```shellinterface
Vlan-interface2
ip address x.x.x.x 255.255.255.0
ip address 192.168.1.2 255.255.255.252 sub
ip policy-based-route aaa
#

acl number 3000
rule 0 permit ip source x.x.x.x 0.0.0.255
#

policy-based-route aaa permit node 2
if-match acl 3000
apply next-hop 192.168.1.1
#
```

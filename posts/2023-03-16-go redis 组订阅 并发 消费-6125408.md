> **内容介绍**
>
> 本文用 Go 演示 Redis Pub/Sub 的"组订阅"与并发消费：
> 生产者协程向 group1 频道 Publish 100 条消息，主协程 Subscribe 订阅 group1，
> 并 PSubscribe 匹配 group1:*，随后启动 10 个 goroutine 从
> pubSub.Channel() 中持续接收打印，用 sync.WaitGroup 等待它们。

> **技术备注**
>
> Pub/Sub 消息"即发即弃"：只投递给发布时在线的订阅者，不落盘、无确认与重试，不适合做可靠队列；
> 可靠消费可改用 List 的 BLPOP，或 Redis 5.0+ 的 Stream（XADD/XREADGROUP，支持消费组与 ACK）。
> pubSub.Channel() 每次调用都会新建一个 Go channel 并启动配套的内部转发 goroutine，应放在循环外只调用一次；
> 本例多个 goroutine 共享其输出，每条消息只会交给其中一个，形成并发分流。
> 代码同时 Subscribe("group1") 与 PSubscribe("group1:*")，但模式 group1:* 匹配不到 group1 本身，且发布只针对 group1，二者取其一即可。
> go-redis 自 v8 起模块路径为 github.com/go-redis/redis/v8，且所有命令需显式传入 context；v9 已迁移至 github.com/redis/go-redis/v9。

## 1. 完整示例代码

```go
package main

import (
    "context"
    "github.com/go-redis/redis/v8"
    "log"
    "sync"
)

var ctx = context.Background()

func ProducerMessageList(rdb *redis.Client, queueListKey string) {
    for i := 0; i < 100; i++ {
        _, err := rdb.Publish(ctx, queueListKey, i).Result()
        if err != nil {
            log.Println(err)
        }
    }
}

func ConsumerMessageList(pubSub *redis.PubSub, node int) {
    for {
        ch := pubSub.Channel()
        for msg := range ch {
            log.Printf("消费者 %d ------------------ channel:%s；message:%s\n", node, msg.Channel, msg.Payload)
        }
    }
}

func main() {

    rdb := redis.NewClient(&redis.Options{
        Addr:     "192.168.117.129:6379",
        Password: "foobared", // no password set
        DB:       0,          // use default DB
    })
    // 测试连接是否成功
    _ = rdb.Ping(context.Background()).Err()

    go ProducerMessageList(rdb, "group1")

    pubSub := rdb.Subscribe(context.Background(), "group1")
    _ = pubSub.PSubscribe(context.Background(), "group1:*")

    var wg sync.WaitGroup
    for i := 0; i < 10; i++ {
        wg.Add(1)
        go ConsumerMessageList(pubSub, i)
    }
    wg.Wait()

}
```

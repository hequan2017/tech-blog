> **About this post**
>
> This post uses Python 3 (urllib + json) to wrap the Zabbix JSON-RPC API: it first
> calls user.login to obtain an auth token, then wraps query functions such as
> hostgroup.get / host.get / item.get / trigger.get. The core is the hand-written
> chaxun(groupid, items_value) — given a group ID and an item key, it first fetches
> all hosts in the group, then filters them one by one for hosts that have the
> target item, and finally queries the latest history value of each host, returning
> a dict mapping "host name: monitored value"; hosts in the group without that item
> are dropped.

> **Technical notes**
>
> Since Zabbix 5.4, the user.login parameter `user` has been renamed to `username`;
> if the call errors on a newer version, change it to `'username': username`. In the
> trigger query it is advisable to change selectItems to `selectItems: ["itemid","key_"]`
> to reduce the returned data. The group ID and key in the example entry
> `chaxun(5, "ProcThreadCount.[2]")` must be replaced with values from your actual
> environment.

---

## 1. How It Works

`chaxun(groupid, items_value)` is a function I wrote myself; you pass in a group ID and an item key, and it returns the hosts together with their values.

A check is performed: if a host does not have the desired item, it is excluded.

## 2. Complete Code

```python
#!/usr/bin/env python
#coding=utf-8
import urllib.request
import json
import re

url = 'http://XXXXXXXXXXXXXXXX/api_jsonrpc.php'
username = 'Admin'
password = 'XXXXX'

# Login
def requestJson(url,values):
    data = json.dumps(values).encode('utf-8')
    req = urllib.request.Request(url, data, {'Content-Type': 'application/json-rpc'})
    response = urllib.request.urlopen(req, data)
    a = response.read().decode(encoding='utf-8')
    output = json.loads(a)
    try:
        message = output['result']
    except:
        message = output['error']['data']
        print(message)
        quit()
    return output['result']

## Login API
def authenticate(url, username, password):
    values = {'jsonrpc': '2.0',
              'method': 'user.login',
              'params': {
                  'user': username,
                  'password': password
              },
              'id': '0'
              }
    idvalue = requestJson(url,values)
    return idvalue

auth=authenticate(url,username,password)

## Query group IDs {'groupid': '8', 'name': 'Switch'}
def groups(auth):
    values = {
            "jsonrpc":"2.0",
            "method":"hostgroup.get",
              "params":{
                    "output":["groupid","name"],
            },
              'auth': auth,
              'id': '1'
              }
    output = requestJson(url,values)
    return output

## Query hosts
def hosts(auth,a):
    values = {
            "jsonrpc":"2.0",
            "method":"host.get",
              "params":{
                    "output":["groupid","name"],
                    "groupids": a,
            },
              'auth': auth,
              'id': '1'
              }
    output = requestJson(url,values)
    return output

## Query host items
def items(auth,a):
    values = {
            "jsonrpc":"2.0",
            "method":"item.get",
              "params":{
                  "output": ["itemids", "key_"],
                  "hostids": a,
                  },
              'auth': auth,
              'id': '1'
              }
    output = requestJson(url,values)
    return output

## Query an item's history value  'lastvalue': '-14760.0000'
def his(auth,itemids):
    values = {
            "jsonrpc":"2.0",
            "method":"item.get",
              "params":{
                    "output": "extend",
                      "history": 0,
                      "itemids": itemids,
                      "sortfield": "itemid",
                      "sortorder": "DESC",
                      "limit": 1
                  },
              'auth': auth,
              'id': '1'
              }
    output = requestJson(url,values)
    return output

## Query trigger values and monitored items
def trigger(auth):
    values = {
              "jsonrpc":"2.0",
              "method": "trigger.get",
              "params": {
                  "output": [
                        "description",
              ],
              "filter": {
                  "hostid": "10108",
              },
              "selectItems":"",
              "sortfield": "hostname",
              "sortorder": "DESC"
              },
              'auth': auth,
              'id': '1'
              }
    output = requestJson(url,values)
    return output

def  chaxun(groupid,items_value):

    host = hosts(auth, groupid) ## get all hosts

    host_name = []
    host_id = []

    for i in host:
        host_id.append(i['hostid'])
        host_name.append(i['name'])
    hostall = dict(zip(host_id, host_name))  ## all hosts     '10118': 'Cloud host: 10.8.8.65','10119': 'Cloud host: 10.8.8.66',

    host_names = []
    host_ids = []
    for i in host_id:    ## filter:   host IDs and names of hosts that have the item and match
        host_id_value = items(auth, i) ## [{'itemid': '29272', 'key_': 'BiosDate'},{'itemid': '29272', 'key_': 'BiosDate'}]
        for index, value in enumerate(host_id_value):
            for k, v in value.items():
                if  items_value  in v:           ## items_value
                    host_ids.append(host_id_value[index]['itemid'])
                    host_names.append(i)



    host_names_list = []
    for i in   host_names:  ## use the matching hosts to get their names
        for k, v in hostall.items():
            if i in k:
                host_names_list.append(v)

    history_value = []
    for i in host_ids:         ## query the latest history value of each host ID
        history = his(auth, i)
        history_value.append(history[0]['lastvalue'])
    host_value = dict(zip(host_names_list, history_value))

    return  host_value  # return the host-to-value mapping

ret = chaxun(5, "ProcThreadCount.[2]")#  group ID and the item to query
print(ret)
```

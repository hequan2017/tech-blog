> **About this post**
>
> Use the AWS SDK for Go to query EC2 instance asset information in a given Region: create a session with static AK/SK credentials, call `DescribeInstances`, iterate over Reservations and Instances, and print key fields such as instance ID, Name tag, and private IP — suitable for collecting cloud host inventories for import into a CMDB.

> **Technical notes**
>
> The example is based on AWS SDK for Go v1 (2020); AWS has since released SDK for Go v2 (`aws-sdk-go-v2`), which is modular and supports IAM role chains and the newer APIs — v2 is recommended for new projects. The import path in the original was truncated; it should actually be `github.com/aws/aws-sdk-go/...`. Hardcoded credentials are only suitable for testing; in production, use an IAM role or environment variables instead.

---

```go
package main

import (
	"fmt"

	"github.com/aws/aws-sdk-go/aws"
	"github.com/aws/aws-sdk-go/aws/credentials"
	"github.com/aws/aws-sdk-go/aws/session"
	"github.com/aws/aws-sdk-go/service/ec2"
)

func GetInstances() (*ec2.DescribeInstancesOutput, error) {
	sess, _ := session.NewSession(&aws.Config{
		Credentials: credentials.NewStaticCredentials("111111111111", "2222222222222222222222222222222", ""),
		Region:      aws.String("ap-east-1"),
	})
	svc := ec2.New(sess)
	result, err := svc.DescribeInstances(nil)

	if err != nil {
		return nil, err
	}
	return result, nil
}

func main() {

	result, err := GetInstances()
	if err != nil {
		fmt.Println("Got an error retrieving information about your Amazon EC2 instances:")
		fmt.Println(err)
		return
	}
	for _, r := range result.Reservations {
		//fmt.Println(r) prints all returned information
		for _, i := range r.Instances {
			fmt.Println("   " + *i.InstanceId + "   " +  *i.Tags[0].Value+"   " + *i.PrivateIpAddress + "   " + * + "   " + *i.InstanceType)
		}
	}
}
```

> **About this post**
>
> This post uses the gin-vue-admin template to demonstrate an "object-oriented" approach to permission control: add a JSON condition field `v3` to the `sys_apis` table, and through a custom Gin middleware, perform field-level filtering on the request body or query parameters of create, query, update, and other requests on top of Casbin, achieving finer-grained data permissions.

> **Technical notes**
>
> The example was written against an early version of gin-vue-admin and Casbin; later versions of gin-vue-admin have adjusted the directory structure and model fields, so the code is intended only as a reference for the idea. The `gofasion` library is rarely used; consider replacing it with a more common JSON parsing solution such as `encoding/json` or `gjson`.

---

### An Object-Oriented Permission Design Example Based on the gin-vue-admin Template

> The template is based on gin-vue-admin

> The main modification is to the sys_apis table: add a field v3 for permission checks, for example: {"name":"test"}

> At its core, this is still an extension built on top of casbin.

### Permission

> Permission :     /dep/getDepList    dep  GET  {"name":"test"}

> The v3 field is mainly the judgment condition. Only requests that match it are allowed through. For example, if you want to create an asset, only those with name equal to test are allowed to pass. Other permission checks work the same way.

### Middleware

> Add this middleware for verification on the modules that need it.

> As for find, update, and delete, they still need to be configured separately for different modules — mainly when it comes to selecting the database.

```go
func CasbinHandlerObject() gin.HandlerFunc {
	return func(c *gin.Context) {
		obj := c.Request.URL.RequestURI()
		act := c.Request.Method
		db := global.GVA_DB.Model(&model.ExaCustomer{})
		obj1 := strings.Split(obj, "?")
		var a model.SysApi
		_ = db.Where(" path = (?)", obj1[0]).Where(" method =  (?)", act).First(&a).Error

		url := strings.Split(obj, "/")
		if url[2][:3] == "cre" {
			type str1 map[string]string
			data1 := str1{}
			if a.V3 == "" { // permission condition
				// means no permission restriction
				c.Next()
			} else {
				if err := json.Unmarshal([]byte(a.V3), &data1); err != nil {
					response.Result(response.ERROR, gin.H{}, "权限设置错误", c)
					c.Abort()
					return
				}
				for k, v := range data1 {
					body := c.Request.Body
					dataByte, _ := ioutil.ReadAll(body)
					fSion := gofasion.NewFasion(string(dataByte))
					name := fSion.Get(k).ValueStr()
					c.Request.Body = ioutil.NopCloser(bytes.NewBuffer(dataByte))
					if name == v {
						c.Next()
						return
					}
				}
				response.Result(response.ERROR, gin.H{}, "权限不足", c)
				c.Abort()
				return
			}

		}
		if url[2][:3] == "get" {
			type str1 map[string]string
			data1 := str1{}
			if a.V3 == "" { // permission condition
				// means no permission restriction
				c.Next()
			} else {
				if err := json.Unmarshal([]byte(a.V3), &data1); err != nil {
					response.Result(response.ERROR, gin.H{}, "权限设置错误", c)
					c.Abort()
					return
				}
				data := c.Request.URL.RawQuery // page=1&pageSize=10
				re := ""
				for k, v := range data1 {
					re = "&" + k + "=" + v
				}
				c.Request.URL.RawQuery = data + re //
				c.Next()

			}
		}

		if url[2][:3] == "upd" { // find and delete are similar to this; both first obtain the id via GET or POST
			id := 36 // com.StrTo(c.Query("id")).MustInt(),  com.StrTo(c.Param("id")).MustInt()
			type str1 map[string]string
			data1 := str1{}
			if a.V3 == "" { // permission condition
				// means no permission restriction
				c.Next()
			} else {
				if err := json.Unmarshal([]byte(a.V3), &data1); err != nil {
					response.Result(response.ERROR, gin.H{}, "权限设置错误", c)
					c.Abort()
					return
				}
				var a model.Dep
				type depStruct struct {
					name string
				}
				dep1 := depStruct{}
				for k, v := range data1 {
					if k == "name" {
						 = v
					}
				}
				err := db.Where("id = ?", id).Where(dep1).First(&a).Error  // you can select different databases based on a.api_group
				if err != nil {
					response.Result(response.ERROR, gin.H{}, "更新权限不足", c)
					c.Abort()
					return
				}
				c.Next()
			}
		}
		c.Next()
	}
}
```

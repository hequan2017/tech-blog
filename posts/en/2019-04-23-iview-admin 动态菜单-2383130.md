> **About this post**
>
> A complete retrofit of iview-admin 2.5.0 to implement backend-driven dynamic menus/dynamic routing: the backend returns the menu JSON, and the frontend adds router-util.js, which filters the menu data and converts it into route objects (component strings into component references), registers them dynamically through vuex's updateMenuList by calling `router.addRoutes`, and calls initRouter on successful login and when the app mounts. This also fixes the bug in the referenced article where the menu did not show on the first login.
>
> **Technical notes**
>
> Applies only to iview-admin 2.5.0 (Vue 2 + vue-router 3); vue-router 4 has removed `addRoutes` (replaced by `addRoute`), and the iview-admin project itself is no longer maintained.

---

### iview-admin 2.5.0 Dynamic Menus

#### Notes

Most of this article is excerpted from https://www.cnblogs.com/smilexumu/p/10521612.html, but that article has a small bug: the menu does not show on the first login — I don't know whether the author left something out or something else was wrong.

This article fixes that bug, and it has been tested and works. It applies only to version 2.5.0; other versions have not been tested.

```javascript
import { getToken, localSave, localRead } from '@/libs/util'
// import config from '@/config'
import { lazyLoadingCop } from '@/libs/tools'
import { getMockMenuData } from '@/api/data'
import Main from '@/components/main' // Main is the layout component; it is not returned by the backend and is imported separately here
import parentView from '@/components/parent-view' // Method for getting components
import store from '@/store' // parentView is the second-level layout component; it is not returned by the backend and is imported separately here
// eslint-disable-next-line no-unused-vars
const _import = require('@/router/_import_' + process.env.NODE_ENV)

var gotRouter
// Initialize the routes
export const initRouter = () => {
  console.log('开始初始化路由')
  if (!getToken()) {
    console.log('没有获取到token')
    return
  }
  //  Async request
  /*  axios.get(baseUrl+'/menuList',{
    header:{'Authorization':getToken()}
  }).then(res=>{
    var menuData=res.data.data;
    localSave('route',JSON.stringify(menuData));
    gotRouter=formatMenu(menuData);
    vm.$store.commit('updateMenuList',gotRouter);
  }); */
  var routerData
  if (!gotRouter) {
    getMockMenuData().then(res => {
      routerData = res.data // routes fetched from the backend
      console.log('保存到本地', routerData)
      localSave('dynamicRouter', JSON.stringify(routerData)) // save the routes to localStorage
      gotRouter = filterAsyncRouter(routerData) // filter the routes and convert route components
      store.commit('updateMenuList', gotRouter)
      dynamicRouterAdd()
    })
  } else {
    gotRouter = dynamicRouterAdd()
  }
  return gotRouter
}

// Load the route menu: read the routes from localStorage, used when creating routes
export const dynamicRouterAdd = () => {
  let dynamicRouter = []
  let data = localRead('dynamicRouter')
  console.log('从本地加载出来', data)
  if (!data) {
    return dynamicRouter
  }
  dynamicRouter = filterAsyncRouter(JSON.parse(data))

  return dynamicRouter
}

// @Function: iterate over the route strings sent from the backend and convert them into component objects
export const filterAsyncRouter = asyncRouterMap => {
  const accessedRouters = asyncRouterMap.filter(route => {
    if (route.component) {
      if (route.component === 'Main') {
        // special handling for the Main component
        route.component = Main
      } else if (route.component === 'parentView') {
        // special handling for the parentView component
        route.component = parentView
      } else {
        // route.component = _import(route.component)
        route.component = lazyLoadingCop(route.component)
      }
    }
    if (route.children && route.children.length) {
      route.children = filterAsyncRouter(route.children)
    }
    return true
  })
  return accessedRouters
}
```

#### Mounting

```javascript

  el: '#app',
  router,
  i18n,
  store,
  render: h => h(App),
  mounted() {
    initRouter() // New: call the method to generate routes dynamically
```

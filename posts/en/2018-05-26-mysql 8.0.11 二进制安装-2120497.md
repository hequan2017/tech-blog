> **About this post**
>
> This post presents a fully automated installation script for the MySQL 8.0.11 binary package (glibc2.12/x86_64): it downloads and extracts the package,
> creates the mysql user plus the data/slow-log directories, and generates /etc/my.cnf (core parameters such as the utf8mb4 character set, a 4G InnoDB
> buffer pool, the slow query log, MIXED binlog, and a server-id derived from the NIC IP). It then initializes the data directory with --initialize-insecure,
> registers the mysqld service with boot-time autostart, sets the root
> password, removes non-local accounts, and finally runs the CheckInstall self-check and prints the port listening status.

> **Technical notes**
>
> CentOS 7 reached end of life (EOL) on June 30, 2024; migrating to Rocky Linux 9 / AlmaLinux 9 / openEuler is recommended.
> MySQL 8.0.11 is an early minor release of the 8.0 series, whose standard support ended in April 2026; for production environments,
> upgrade to the latest 8.0 patch release or migrate to 8.4 LTS.
> MySQL 8.0 changed the default authentication plugin to caching_sha2_password. If older clients/drivers fail to connect, add
> default_authentication_plugin=mysql_native_password to the [mysqld] section for compatibility.

The script below is split into sections in execution order; concatenating the code blocks of each section in turn yields the complete script.

## 1. Preparation: download the binary package and install dependencies

```shell
#!/bin/bash

```

## 3. InitInstall(): system initialization

```shell
# Initialize  the installation related content.
function InitInstall()
{
    cat /etc/issue
    uname -a
    MemTotal=`free -m | grep Mem | awk '{print  $2}'`
    echo -e "\n Memory is: ${MemTotal} MB "
    #Set timezone
    rm -rf /etc/localtime
    ln -s /usr/share/zoneinfo/Asia/Shanghai /etc/localtime
    #Delete Old Mysql program
    rpm -qa|grep mysql
    rpm -e mysql
    #Disable SeLinux
    if [ -s /etc/selinux/config ]; then
        sed -i 's/SELINUX=enforcing/SELINUX=disabled/g' /etc/selinux/config
    fi
    setenforce 0
}
```

## 4. InstallDependsAndOpt(): optimize system resource limits

```shell
#Installation of depend on and optimization options.
function InstallDependsAndOpt()
{
cd $cur_dir
cat >>/etc/security/limits.conf<<EOF
* soft nproc 65535
* hard nproc 65535
* soft nofile 65535
* hard nofile 65535
EOF
}
```

## 5. InstallMySQL8011(): install the MySQL core

Backs up the old my.cnf, extracts the binary package to /data/mysql8011, and creates the mysql user and related directories:

```shell
#Install MySQL
function InstallMySQL8011()

{
echo "============================Install MySQL 8.0.11=================================="
cd $cur_dir
#Backup old my.cnf
#rm -f /etc/my.cnf
if [ -s /etc/my.cnf ]; then
    mv /etc/my.cnf /etc/my.cnf.`date +%Y%m%d%H%M%S`.bak
fi

echo "============================MySQL 8.0.11 installing…………========================="

#mysql directory configuration
cd   /data/src/
tar  xvf   mysql-8.0.11-linux-glibc2.12-x86_64.tar.gz
mv mysql-8.0.11-linux-glibc2.12-x86_64   /data/mysql8011

groupadd mysql -g 512
useradd -u 512 -g mysql -s /sbin/nologin -d /home/mysql mysql

mkdir -p /data/mysqldata
mkdir -p /data/slowlog
chown -R mysql:mysql /data/mysqldata
chown -R mysql:mysql /data/mysql8011
chown -R mysql:mysql /data/slowlog
```

Generates /etc/my.cnf (the server-id is taken from the last two octets of the eth0 NIC IP):

```shell
#edit /etc/my.cnf

```

Initializes the data directory, registers the service, and starts it:

```shell
/data/mysql8011/bin/mysqld     --defaults-file=/etc/my.cnf     --user=mysql --datadir=/data/mysqldata    --basedir=/data/mysql8011  --initialize-insecure

cp  /data/mysql8011/support-files/mysql.server   /etc/init.d/mysqld

chmod    700 /etc/init.d/mysqld

chkconfig --add mysqld
chkconfig --level 2345 mysqld on

cat >> /etc/ld.so.conf.d/mysql-x86_64.conf<<EOF
/data/mysql8011/lib
EOF
ldconfig

if [ -d "/proc/vz" ];then
    ulimit -s unlimited
fi

/etc/init.d/mysqld start
cat >> /etc/profile <<EOF
export PATH=$PATH:/data/mysql8011/bin
export LD_LIBRARY_PATH=$LD_LIBRARY_PATH:/data/mysql8011/lib
EOF

source  /etc/profile
```

Sets the root password and removes non-local accounts:

```shell
/data/mysql8011/bin/mysqladmin -u root password $mysqlrootpwd

cat > /tmp/mysql_sec_script<<EOF
use mysql;
delete from mysql.user where user!='root' or host!='localhost';
flush privileges;
EOF

/data/mysql8011/bin/mysql -u root -p$mysqlrootpwd -h 127.0.0.1   < /tmp/mysql_sec_script
rm -f /tmp/mysql_sec_script
#/etc/init.d/mysqld restart

echo "============================MySQL 8.0.11 install completed========================="
}
```

## 6. CheckInstall(): post-installation self-check

```shell
function CheckInstall()
{
echo "===================================== Check install ==================================="
clear
ismysql=""
echo "Checking..."
if [ -s /data/mysql8011/bin/mysql ] && [ -s /data/mysql8011/bin/mysqld_safe ] && [ -s /etc/my.cnf ]; then
  echo "MySQL: OK"
  ismysql="ok"
  else
  echo "Error: /data/mysql8011 not found!!!MySQL install failed."
fi
if [ "$ismysql" = "ok" ]; then

echo "Install MySQL 8.0.11 completed! enjoy it."
echo "========================================================================="

netstat -lntup
else
echo "Sorry,Failed to install MySQL!"
echo "You can tail /root/mysql-install.log from your server."
fi
}
```

## 7. Run the installation

```shell
#The installation log

InitInstall 2>&1 | tee /root/mysql-install.log
InstallDependsAndOpt 2>&1 | tee -a /root/mysql-install.log
InstallMySQL8011 > /dev/null
CheckInstall 2>&1 | tee -a /root/mysql-install.log
source  /etc/profile
```

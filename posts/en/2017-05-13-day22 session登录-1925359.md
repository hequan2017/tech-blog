a. How Session works:

- A session is a key-value pair stored on the server

Set the default operations in the settings file (common configuration):

```python
SESSION_COOKIE_NAME = "sessionid"         # Key under which the session cookie is saved in the browser, i.e.: sessionid=<random string> (default)
SESSION_COOKIE_PATH = "/"                 # Path where the session cookie is saved (default)
SESSION_COOKIE_DOMAIN = None              # Domain for the session cookie (default)
SESSION_COOKIE_SECURE = False             # Whether the cookie is sent over HTTPS only (default)
SESSION_COOKIE_HTTPONLY = True            # Whether the session cookie supports HTTP transport only (default)
SESSION_COOKIE_AGE = 1209600              # Age of the session cookie in seconds (2 weeks) (default)
SESSION_EXPIRE_AT_BROWSER_CLOSE = False   # Whether closing the browser expires the session (default)
# set_cookie('k', 123)
SESSION_SAVE_EVERY_REQUEST = False        # Whether to save the session on every request; by default it is saved only when modified (default)
```

## 9. Attaching the CSRF Token Globally in Ajax (jquery.cookie.js)

All AJAX requests:

```html
<script src="/static/plugin/jquery/jquery-1.8.0.js"></script>
<script src="/static/plugin/jquery/jquery.cookie.js"></script>
<script type="text/javascript">
var csrftoken = $.cookie('csrftoken');

function csrfSafeMethod(method) {
    // these HTTP methods do not require CSRF protection
    return (/^(GET|HEAD|OPTIONS|TRACE)$/.test(method));
}

$.ajaxSetup({
    beforeSend: function(xhr, settings) {
        if (!csrfSafeMethod(settings.type) && !this.crossDomain) {
            xhr.setRequestHeader("X-CSRFToken", csrftoken);
        }
    }
});

function Do(){
    $.ajax({
        url: "/app01/test/",
        data: {id: 1},
        type: 'POST',
        success: function(data){
            console.log(data);
        }
    });
}
</script>
```

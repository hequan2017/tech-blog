> **About this post**
>
> A simple SSH client wrapped with Go's golang.org/x/crypto/ssh package: it establishes a connection using password authentication, runs a remote command, and captures both standard output and standard error at once via `CombinedOutput`. A solid skeleton for small batch-ops tools.
>
> **Technical notes**
>
> In production, do not skip host key verification by making `HostKeyCallback` return nil (it exposes you to man-in-the-middle attacks). Newer versions of x/crypto/ssh provide `gossh.InsecureIgnoreHostKey()` for testing only; for real use, pair it with knownhosts verification, and prefer key-based authentication in sensitive environments.

### Running Commands over SSH in Go

```go
package main

import (
   "fmt"
   "net"

   gossh "golang.org/x/crypto/ssh"
)

type Cli struct {
   user    string
   pwd     string
   addr    string
   client  *gossh.Client
   session *gossh.Session
   LastResult string
}

func (c *Cli) Connect() (*Cli, error) {
   config := &gossh.ClientConfig{}
   config.SetDefaults()
   config.User = c.user
   config.Auth = []gossh.AuthMethod{gossh.Password(c.pwd)}
   config.HostKeyCallback = func(hostname string, remote net.Addr, key gossh.PublicKey) error { return nil }
   client, err := gossh.Dial("tcp", c.addr, config)
   if nil != err {
      return c, err
   }
   c.client = client
   return c, nil
}

func (c Cli) Run(shell string) (string, error) {
   if c.client == nil {
      if _,err := c.Connect(); err != nil {
         return "", err
      }
   }
   session, err := c.client.NewSession()
   if err != nil {
      return "", err
   }
   defer session.Close()
   buf, err := session.CombinedOutput(shell)

   c.LastResult = string(buf)
   return c.LastResult, err
}

func main() {
   cli := Cli{
      user:"root",
      pwd: "xxxxxxxxxxxxxxxxxxxxx",
      addr: "192.168.100.50:22",
   }
   output, err := cli.Run("pwd")
   fmt.Printf("%v\n%v", output, err)
}
```

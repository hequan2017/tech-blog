> **About this post**
>
> Use a Jenkins Pipeline to deploy and roll back a Java JAR package: the `active choices` plugin provides cascading parameters, and when `rollback` is selected it dynamically lists the historical versions in the backup directory. The Pipeline runs in stages — "generate version number → pull code → Maven build → upload → back up → stop/start service" — and sends a DingTalk robot notification on success.

> **Technical notes**
>
> The examples are based on Jenkins 2.x Pipeline and the `active choices` plugin; the DingTalk notification section of the script contains a truncated `${}` placeholder that must be completed with a real variable such as `${BUILD_URL}`. From Jenkins 2.462+ onward, plugin installation and security policies have changed, so verify in a test environment first.

### Jenkins Pipeline for Deploying and Rolling Back JAR Packages

> Plugin: active choices

![](../assets/2543374/01_e05cc494fe1be116e12b083dacfbe291.png)

```groovy
properties ([
    parameters([
        [$class: 'CascadeChoiceParameter',
            choiceType: 'PT_SINGLE_SELECT',
            description: 'Select when rolling back',
            referencedParameters: "action",
            filterLength: 1,
            name: 'rollbck',
            script: [
                $class: 'GroovyScript',
                script: [
                    classpath: [],
                    sandbox: false,
                    script: '''path="/data/jenkins/workspace/test/backup/"
rollback=['bash', '-c', "ls -t1 ${path} "].execute().text.readLines()
if (action.equals("rollback")) {
  return rollback
} else {
  return ["Displayed after selecting rollback"]
}'''

                ]
            ]
        ]
    ])
])

pipeline {

    agent any

    options {
        buildDiscarder(logRotator(numToKeepStr: '5'))
    }


    environment {
	 PROJECT='test'
     GIT_URL='https://gitee.com/test/test.git'
	 DIR_WORK="/data/work-test/${PROJECT}"
	 TARGET_DIR="/data/test/${PROJECT}"
     START="sh ./test.sh start"
     STOP="sh ./test.sh stop"
    }

    parameters {
        string(name: 'ENV', defaultValue: 'test', description: 'Environment')
        string(name: 'BRANCH', defaultValue: 'master', description: 'Branch')
        choice(name: 'SERVICE', choices: ['test',], description: 'test')
        choice(name: 'action', choices: ['deploy','rollback'], description: 'deploy build     rollback roll back')
    }

    stages {
       stage('Generate version number'){
            steps{
                script{
					DATA_BASE ="${sh(script:'date +%Y%m%d_%H%M%S | xargs echo -n', returnStdout: true)}"
					env.DATA_BASE = DATA_BASE
					echo "/data/backup/${PROJECT}/${DATA_BASE}"
				    sh "ssh root@${SERVICE}    mkdir -p ${DIR_WORK}/${DATA_BASE}"
                    sh "mkdir -p   ${WORKSPACE}/backup"
                    sh "printenv"
					}
				}
            }
       stage ('Pull code') {
         when { expression { return params.action  != 'rollback' } }
         steps {
			git branch: "${BRANCH}",credentialsId: 'test',url: "${GIT_URL}"
		}
	   }
	   stage("Build package"){
    	 when { expression { return params.action  != 'rollback' } }
    		steps {
            sh '/usr/local/maven/bin/mvn -s /usr/local/maven/conf/settings.xml -e clean package  -U -Ptest'
		    sh "cp target/*.jar  ${WORKSPACE}/backup/${BUILD_ID}_${DATA_BASE}_${PROJECT}_test.jar"
		    sh '''cd ${WORKSPACE}/backup/
		    total_file_num=`ls -l | grep -v 'total' | wc -l`
            if [ $total_file_num -gt 5 ];
            then
                num=`expr $total_file_num - 5`
                files=`ls -ltr | grep -v 'total' | awk '{print $9}' | head -n $num`
                echo $files | xargs rm -f
                echo "Keep 5 backups, delete older ones: " $files
            fi
		    '''
       }
	   }
	    stage("Upload package [deploy]"){
	   when { expression { return params.action  != 'rollback' } }
		steps {
		sh "scp target/*.jar   root@${SERVICE}:${DIR_WORK}/${DATA_BASE}/${PROJECT}.jar "
		sh "rm -rf target/*.jar "
		}
      }
      stage("Upload package [rollback]"){
       when { expression { return params.action  == 'rollback' } }
		steps {
		sh "scp ${WORKSPACE}/backup/${rollbck}   root@${SERVICE}:${DIR_WORK}/${DATA_BASE}/${PROJECT}.jar  "
		}
      }
	  stage("Create backup directory"){
		steps {
            sh "ssh root@${SERVICE}   mkdir  -p  /data/backup/${PROJECT}/${DATA_BASE}"
		}
      }
      stage("Stop service"){
		steps {
        sh 'ssh root@${SERVICE}   "cd ${TARGET_DIR} && ${STOP}"'
		}
      }
      stage("Back up package"){
		steps {
        sh "ssh root@${SERVICE}    mv ${TARGET_DIR}/${PROJECT}.jar  /data/backup/${PROJECT}/${DATA_BASE}"
		}
      }
	  stage("Replace package"){
		steps {
        sh 'ssh root@${SERVICE}     mv  ${DIR_WORK}/${DATA_BASE}/${PROJECT}.jar   ${TARGET_DIR}'
		}
      }
	 stage("Start service"){
		steps {
		    withEnv(['JENKINS_NODE_COOKIE=dontKillMe']) {
                sh 'ssh root@${SERVICE}    "cd ${TARGET_DIR} && ${START}"'
		    	sh 'ssh root@${SERVICE}    "ps -aux | grep ${PROJECT}.jar | grep -v grep " '
		    	script{
    		    	re ="${sh(script:'ssh root@${SERVICE}    ps -aux | grep ${PROJECT}.jar | grep -v grep |wc -l | xargs echo -n ', returnStdout: true)}"
    		    	echo re
    				if (re == '1' ) {
    				  	="Service started successfully"
    				}else{
    				   	="Service failed to start"
    				}
		       }
		    }
		}
	    post {
            success {
                    dingtalk (
                        robot: 'test',
                        type: 'TEXT',
                        text: [
                            "${PROJECT} ${}",
                        ]
                    )
                }
            }
		}
    }
}
```

> **About this post**
>
> A cron job service implemented in Go that supports dynamically adding, removing, starting and stopping jobs through an HTTP API: jobs are stored in MySQL (the go_cron table), scheduled with gcron from the gogf framework, and Gin exposes the /crons/:id endpoints -- GET (start/stop), POST (create), PUT (update), and DELETE (remove). Each job sends a shell command to a specified intranet IP host.
>
> **Technical notes**
>
> The early gogf framework APIs used in this post (the gcron/glog package paths, and gorm v1's jinzhu/gorm import path) have since changed considerably: GoFrame is now at v2 (github.com/gogf/gf/v2), and gorm has moved to gorm.io/gorm. The code is intended only as a reference for the approach; running it as-is would require substantial adjustments.
>
> CentOS 7 reached end of maintenance (EOL) on June 30, 2024; hosts still based on it should migrate to Rocky Linux 9 / AlmaLinux 9 or openEuler.

### Go Dynamic Cron Job API

#### Usage

- Connect to the database table first, then start the service
- Initialize all cron jobs, then modify them dynamically through the API.

Database table:

```sql
CREATE TABLE `go_cron` (
  `id` int(10) unsigned NOT NULL AUTO_INCREMENT,
  `time` varchar(100) DEFAULT '' COMMENT 'Cron time',
  `ip` varchar(100) DEFAULT '' COMMENT 'Intranet IP',
  `shell` varchar(50) DEFAULT '' COMMENT 'Command',
  `env` varchar(50) DEFAULT '' COMMENT 'Environment',
  `run` int(10)  DEFAULT '0' COMMENT 'Whether to run',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=17 DEFAULT CHARSET=utf8;
```

> This example only executes commands on IP hosts; adapt it to your own needs for other use cases.

#### Code

```go
package main

import (
	"fmt"
	"strconv"

	"github.com/Unknwon/com"
	"github.com/gin-gonic/gin"
	"github.com/gogf/gf/g/os/gcron"
	"github.com/gogf/gf/g/os/glog"
	"github.com/jinzhu/gorm"
	_ "github.com/jinzhu/gorm/dialects/mysql"
)

var db *gorm.DB

type Cron struct {
	ID        int    `gorm:"primary_key" json:"id"`
	Time      string `json:"time"`
	Ip        string `json:"ip"`
	Shell string `json:"shell"`
	Env       string `json:"env"`
	Run       int    `json:"run"`
}

func GetCron(id int) (user Cron) {
	db.Where("id = ?", id).First(&user)
	return
}

func GetCronAll() ([] *Cron) {
	var cron []*Cron
	db.Where("run = ?", 0).Find(&cron)
	return cron
}

func InitDb() {
	var err error
	db, err = gorm.Open("mysql", fmt.Sprintf("root:123456@tcp(192.168.100.50)/go?charset=utf8&parseTime=True&loc=Local", ))

	if err != nil {
		glog.Println(err)
	}

	gorm.DefaultTableNameHandler = func(db *gorm.DB, defaultTableName string) string {
		return "go_" + defaultTableName
	}

	db.SingularTable(true)
	db.DB().SetMaxIdleConns(10)
	db.DB().SetMaxOpenConns(100)
}

func main() {
	InitDb()

	r := gin.Default()

	gc := gcron.New()
	data := GetCronAll()

	for _, v := range data {
		_, err := gc.Add(v.Time, func() { glog.Println(v.Shell) }, strconv.Itoa(v.ID))
		if err != nil {
			fmt.Println("Error initializing cron jobs ",err)
		}
	}

	r.GET("/crons/:id", func(c *gin.Context) {
		id := com.StrTo(c.Param("id")).MustInt()
		run := com.StrTo(c.Query("run")).MustInt()
		if run == 0 {
			gc.Start(strconv.Itoa(id))
			glog.Println("Start", id)
			c.JSON(200, gin.H{
				"data": 0,
			})
		} else if run == 1 {
			glog.Println("Stop", id)
			gc.Stop(strconv.Itoa(id))
			c.JSON(200, gin.H{
				"data": 1,
			})
		} else {
			glog.Println("Invalid parameter", id)
			c.JSON(400, gin.H{
				"data": "Invalid parameter, use run=0/1",
			})
		}
	})

	r.POST("/crons/:id", func(c *gin.Context) {
		id := com.StrTo(c.Param("id")).MustInt()
		data := GetCron(id)
		_, err := gc.Add(data.Time, func() { glog.Println(data.Shell) }, strconv.Itoa(data.ID))
		if err != nil {
			glog.Println("Error adding job ",err)
			c.JSON(400, gin.H{
				"data":err,
			})
		}else {
			glog.Println("Job added successfully ")
			c.JSON(200, gin.H{
				"data":0,
			})
		}
	})

	r.PUT("/crons/:id", func(c *gin.Context) {
		id := com.StrTo(c.Param("id")).MustInt()
		data := GetCron(id)
		gc.Stop(strconv.Itoa(id))
		gc.Remove(strconv.Itoa(id))
		_, err := gc.Add(data.Time, func() { glog.Println(data.Shell) }, strconv.Itoa(data.ID))
		if err != nil {
			glog.Println("Error updating job ",err)
			c.JSON(400, gin.H{
				"data":err,
			})
		}else {
			glog.Println("Job updated successfully ")
			c.JSON(200, gin.H{
				"data":0,
			})
		}
	})

	r.DELETE("/crons/:id", func (c *gin.Context) {
		id := com.StrTo(c.Param("id")).MustInt()

		fmt.Println(gc.Search(strconv.Itoa(id)))
		glog.Println("Delete", id)
		gc.Stop(strconv.Itoa(id))
		fmt.Println(gc.Search(strconv.Itoa(id)))
		gc.Remove(strconv.Itoa(id))
		fmt.Println(gc.Search(strconv.Itoa(id)))

		err := gc.Search(strconv.Itoa(id))
		if err != nil {
			c.JSON(400, gin.H{
				"data": "Deleted successfully",
			})
		}else {
			c.JSON(200, gin.H{
				"data": 0,
			})
		}

	})
	_ = r.Run("127.0.0.1:8080")
}
```

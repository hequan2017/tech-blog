> **About this post**
>
> A collection of examples for calling the Zabbix JSON-RPC API with Python 3's native urllib: it wraps a `requestJson` method for sending requests, demonstrates authenticating with user.login to obtain auth, querying host groups with hostgroup.get, hosts with host.get, monitored items with item.get, and triggers with trigger.get, plus a small end-to-end example that looks up a host's triggers and then fetches their latest values.

> **Technical notes**
>
> The Zabbix API's JSON-RPC 2.0 protocol has remained stable to this day; methods such as `user.login` and `hostgroup.get` still work in recent versions, but some field names have changed (for example, since 6.0 the `itemids` field in `item.get`'s `output` should be `itemid`, and the `selectItems` parameter must be declared explicitly). It is advisable to consult the official API documentation for your current Zabbix version.

import urllib.request

# Log in
def requestJson(url, values):
    data = json.dumps(values).encode('utf-8')
    req = urllib.request.Request(url, data, {'Content-Type': 'application/json-rpc'})
    response = urllib.request.urlopen(req, data)
    a = response.read().decode(encoding='utf-8')
    output = json.loads(a)
    #    print output
    try:
        message = output['result']
    except:
        message = output['error']['data']
        print(message)
        quit()

    return output['result']

## Login API
def authenticate(url, username, password):
    values = {'jsonrpc': '2.0',
              'method': 'user.login',
              'params': {
                  'user': username,
                  'password': password
              },
              'id': '0'
              }
    idvalue = requestJson(url, values)
    return idvalue

# Value of auth
auth = authenticate(url, username, password)

## Query group IDs {'groupid': '8', 'name': 'Switch'}
def groups(auth):
    values = {

        "method": "hostgroup.get",
        "params": {
            "output": ["groupid", "name"],
        },
        'auth': auth,
        'id': '1'
    }
    output = requestJson(url, values)
    return output

        "jsonrpc": "2.0",
        "method": "item.get",
        "params": {
            "output": "extend",
            "history": 0,
            "itemids": itemids,
            "sortfield": "itemid",
            "sortorder": "DESC",
            "limit": 1
        },
        'auth': auth,
        'id': '1'
    }
    output = requestJson(url, values)

## Query trigger values and monitored items  {'description': 'xxxxxxxxxxxxxxx', 'hostname': 'xxxxxxxxxxxxxxx', 'items': [{'itemid': '26399'}], 'triggerid': '17030'}
def trigger(auth, hostid):
    values = {
        "jsonrpc": "2.0",
        "method": "trigger.get",
        "params": {
            "output": [
                "description",
            ],
            "filter": {
                "hostid": hostid,
            },
            "selectItems": "",
            "sortfield": "hostname",
            "sortorder": "DESC"
        },
        'auth': auth,
        'id': '1'
    }
    output = requestJson(url, values)
    return output

> **About this post**
>
> Implementing the singly linked list data structure from scratch in Go: it defines a LinkNode node and a LinkNoder interface, and implements tail append, delete/insert by index, length counting, search, fetching all elements, plus reversing the list both iteratively and recursively. Along the way it also explains the difference between Go methods and functions. Good hands-on practice for data structures and pointer manipulation.
>
> **Technical notes**
>
> After generics were introduced in Go 1.18, this style of "using interface{} as an arbitrary type" can be refactored with `[T any]` for better type safety. The `Delete` method also has an ordering bug: it unlinks the node first and takes the value afterwards (`point.Next` has already been set to next.next, so reading `point.Next.Payload` yields the value after the deleted node) — beware of this in real use.

---

```go
package main

import (
	"fmt"
	"log"
)

// Item can be understood as a generic: it can be any data type
type Item interface {

}

// A node must, besides its own data, point to the next node; the tail node points to nil
type LinkNode struct {
	Payload Item    //  Payload can be any data type
	Next *LinkNode
}

type LinkNoder interface {
	// A Go interface: inside it we can define a series of methods.
	Add(payload Item)
	Delete(index int) Item
	Insert(index int, payload Item)
	GetLength() int
	Search(payload Item) int
	GetAll(index int) Item
	Reverse() *LinkNode
}

// A Go method. Compared with NewLinkNode below, a method can be seen as an object method in OOP:
// although it works like a function, it is bound to an object — the Add here is bound to the head
// LinkNode object. This design sets Go apart from other languages and is a very important part of Go.
func (head *LinkNode) Add(payload Item)  {
	// Add an element to the list by inserting at the tail
	point := head

	for point.Next != nil{
		point = point.Next
	}
	newNode := LinkNode{payload, nil}
	point.Next = &newNode

	// Insert at the head
	//newNode := LinkNode{payload, nil}
	//newNode.Next = head
}

// Delete an element and return the value of the deleted node
func (head *LinkNode) Delete(index int) Item  {
	// Boundary check
	linkLength := head.GetLength()

	if index < 0 || index > linkLength{
		fmt.Printf("index out of range %d, please check.", linkLength)
		return "index out of range, please check."
	}

	point := head
	for i := 0; i<index; i++{
		point = point.Next     // Move point to the index position, then point its next to next.next
	}

	point.Next = point.Next.Next

	data := point.Next.Payload
	return data
}

// Insert an element
func (head *LinkNode) Insert(index int, payload Item){
	linkLength := head.GetLength()

	if index < 0 || index > linkLength{
		fmt.Printf("index out of range %d, please", linkLength)
		return
	}

	point := head
	for i:=0; i<index; i++{
		point = point.Next
	}

	newNode := LinkNode{Payload:payload}
	newNode.Next = point.Next
	point.Next = &newNode
}

func (head *LinkNode) Search(payload Item) int {
	point := head

	index := 0

	for point.Next != nil{
		if point.Payload == payload{
			return index
		}else {
			index ++
			point = point.Next

			// Boundary check
			if index > head.GetLength() -1 {
				break
			}

			continue
		}
	}

	// Check whether the last element matches
	if point.Payload == payload {
		return index
	}

	return -1   // Returned when the element does not exist
}

func (head *LinkNode) GetLength() int  {
	iterator := head

	var length int
	for iterator.Next != nil{
		length ++
		iterator = iterator.Next
	}
	return length
}

func (head *LinkNode) GetAll() []Item  {
	dataList := make([]Item, 0, head.GetLength())

	point := head
	for point.Next != nil{
		dataList = append(dataList, point.Payload)
		point = point.Next
	}

	dataList = append(dataList, point.Payload)
	return dataList
}

// Create a new linked list. A function: compared with the methods above, a function is bound to no object.
// Go functions must declare their parameters and return values; here the parameter is length and the
// return value is a LinkNode object. Apart from the binding, functions and methods are no different.
func NewLinkNode(length int) *LinkNode  {
	if length <= 0{
		fmt.Printf("list length must be greater than 0")
		log.Panic("list length must be greater than 0")
	}
	var head *LinkNode

	head = &LinkNode{}

	for i := 0; i<length; i++{
		var newNode *LinkNode
		newNode = &LinkNode{Payload: i}
		newNode.Next = head
		head = newNode
	}
	return head
}

// Reverse a linked list iteratively
func (head *LinkNode) Reverse() *LinkNode  {
	if head == nil || head.Next == nil{
		return head
	}

	var reverseHead *LinkNode
	var p *LinkNode

	reverseHead = head
	head = head.Next

	reverseHead.Next = nil

	p = head.Next

	for head != nil{
		head.Next = reverseHead
		reverseHead = head

		head = p
		if p != nil{
			p = p.Next
		}
	}

	return reverseHead
}

// Reverse a linked list recursively
func (head *LinkNode) RecursiveReverse() *LinkNode  {
	if head == nil || head.Next == nil{
		return head
	}

	second := head.Next
	newHead := second.RecursiveReverse()

	second.Next = head
	head.Next = nil
	return newHead
}

func main()  {
	t := LinkNode{Payload:1,Next:nil}
	t.Add("123")
	fmt.Println(t.GetLength())
}
```

> **About this post**
>
> This post describes how to use the gitee.com/konyshe/gogo framework to generate CRUD RESTful APIs in one step directly from database tables, without writing any models files. It provides a complete demo (POST/DELETE/PUT/GET routes operating on tables directly) and documents the fix for a Scan error in gogo's source file SQLUtils.go when an INT field is NULL (which requires handling with sql.NullInt64).
>
> **Technical notes**
>
> gogo is a rather obscure personal framework that has not been updated for years; for production projects, consider using an actively maintained combination such as Gin/Echo + GORM/Ent instead. The approach of "directly concatenating the id into the query condition" shown in this post is vulnerable to SQL injection; always use parameterized queries in practice.

---

### One-Click Generation of Backend RESTful APIs

#### Description

This mainly uses the `gitee.com/konyshe/gogo` library; for details, you can check its documentation.

As long as you have connected to the database tables, you can generate RESTful APIs in a simple way without writing any database models files.

The original project has a small issue, which requires modifying the source file SQLUtils.go:

Modification at line 239:

```go
case "INT":
{
    queryData[queryCount][a] = new(sql.NullInt64)
}
```

Addition at line 290:

```go
case *sql.NullInt64:
      result[columnsType[a].Name()] = *s
```

The reason is that int fields in the database allow NULL and default to null, which causes an error when reading:

```text
sql: Scan error on column index 3, name "created_on": converting driver.Value type <nil> ("<nil>") to a int32: invalid syntax
```

#### demo

```go
package main

import (
	"net/http"
	"strconv"

	"gitee.com/konyshe/gogo"
)

func main() {

	// Initialize the database connection
	if err := gogo.SQLInit("mysql", "root:1234567890@tcp(192.168.100.50:3306)/go?charset=utf8&parseTime=true", 10, 1); err != nil {
		gogo.Log().Error(err.Error())
		return
	}

	// Create
	gogo.POST("/restful/:tablename", func(ctx *gogo.HTTPContext) {
		affect, err := gogo.SQLInsert(
			ctx.GetPathParam(":tablename"),
			ctx.GetPostBody())

		if err != nil {
			gogo.Log().Error(err.Error())
			ctx.WriteString(err.Error())
		} else {
			ctx.WriteString(strconv.FormatInt(affect, 10))
		}
	})

	// Delete
	gogo.DELETE("/restful/:tablename/:id", func(ctx *gogo.HTTPContext) {
		affect, err := gogo.SQLDelete(
			ctx.GetPathParam(":tablename"),
			"id="+ctx.GetPathParam(":id"))

		if err != nil {
			gogo.Log().Error(err.Error())
			ctx.WriteString(err.Error())
		} else {
			ctx.WriteString(strconv.FormatInt(affect, 10))
		}
	})

	// Update
	gogo.PUT("/restful/:tablename/:id", func(ctx *gogo.HTTPContext) {
		affect, err := gogo.SQLUpdate(
			ctx.GetPathParam(":tablename"),
			"id="+ctx.GetPathParam(":id"),
			ctx.GetPostBody())

		if err != nil {
			gogo.Log().Error(err.Error())
			ctx.WriteString(err.Error())
		} else {
			ctx.WriteString(strconv.FormatInt(affect, 10))
		}
	})

	// Read
	gogo.GET("/restful/:tablename/:id", func(ctx *gogo.HTTPContext) {
		queryData, err := gogo.SQLQueryByMap(
			"",
			ctx.GetString("feilds"),
			ctx.GetPathParam(":tablename"),
			"id="+ctx.GetPathParam(":id"),
			"", 0, 1)

		if err != nil {
			gogo.Log().Error(err.Error())
			ctx.WriteString(err.Error())
		} else {
			ctx.WriteJSON(queryData)
		}
	})

	// Read
	gogo.GET("/restful/:tablename", func(ctx *gogo.HTTPContext) {
		queryData, err := gogo.SQLQueryByMap(
			ctx.GetString("columnname"),
			ctx.GetString("feilds"),
			ctx.GetPathParam(":tablename"),
			ctx.GetString("where"),
			ctx.GetString("order"),
			ctx.GetInt("offset", 0),
			ctx.GetInt("count", 10))

		if err != nil {
			gogo.Log().Error(err.Error())
			ctx.WriteString(err.Error())
		} else {
			ctx.WriteJSON(queryData)
		}
	})

	// 404 page
	gogo.STATUS(http.StatusNotFound, func(ctx *gogo.HTTPContext) {
		ctx.WriteHeader(http.StatusNotFound)

		l, err := gogo.SQLQueryByMap("webname", "*", "dede_flink", "", "", 0, 100)
		if err == nil {
			ctx.WriteExecute(l,
				"views/notfound.tmpl")
			return
		}
	})

	// Start the HTTP server
	gogo.Run(":8080")
}
```

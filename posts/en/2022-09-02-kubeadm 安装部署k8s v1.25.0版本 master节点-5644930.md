> **About this post**
>
> This post records the complete process of deploying a single-master Kubernetes v1.25.0 cluster on CentOS 8.5 with kubeadm, covering kernel parameter tuning (sysctl), containerd runtime configuration (SystemdCgroup, registry mirror), preparation for kube-proxy in IPVS mode, kubeadm init/join initialization, and the installation and verification of the Calico network plugin.
>
> **Technical notes**
>
> 1. CentOS 8 reached end of maintenance at the end of 2021 and the mirror repositories used in this post are offline; for production environments, consider Rocky Linux 9 / AlmaLinux 9 or openEuler instead;
> 2. Kubernetes v1.25.0 is no longer maintained; upgrade to the current stable release (v1.30+). Newer versions removed support for the Docker shim, so containerd must be used directly;
> 3. The `nf_conntrack_ipv4` kernel module has been renamed to `nf_conntrack` in newer kernels; adjust accordingly if loading fails.

```shell
Preparation:

CentOS Linux release 8.5.2111
uname -a
Linux k8s-master 4.18.0-348.7.1.el8_5.x86_64 #1 SMP Wed Dec 22 13:25:12 UTC 2021 x86_64 x86_64 x86_64 GNU/Linux

hostnamectl set-hostname k8s-master
hostnamectl set-hostname k8s-node1

cat >> /etc/hosts << EOF
192.168.209.128 k8s-master
192.168.209.133 k8s-node1
EOF

yum -y install wget jq psmisc vim net-tools nfs-utils telnet yum-utils device-mapper-persistent-data lvm2 git network-scripts tar curl -y
systemctl stop firewalld && systemctl disable firewalld
systemctl stop iptables && systemctl disable iptables

timedatectl set-timezone Asia/Shanghai
timedatectl set-local-rtc 0
systemctl restart rsyslog
systemctl restart crond

cat > kubernetes.conf<<EOF
# Enable bridge mode [important]
net.bridge.bridge-nf-call-iptables=1
# Enable bridge mode [important]
net.bridge.bridge-nf-call-ip6tables=1
net.ipv4.ip_forward=1
net.ipv4.tcp_tw_recycle=0
# Disable swap; allow it only when the system hits OOM
vm.swappiness=0
# Do not check whether physical memory is sufficient
vm.overcommit_memory=1
# Enable OOM
vm.panic_on_oom=0
fs.inotify.max_user_instances=8192
fs.inotify.max_user_watches=1048576
fs.file-max=52706963
fs.nr_open=52706963
# Disable IPv6 [important]
net.ipv6.conf.all.disable_ipv6=1
net.netfilter.nf_conntrack_max=2310720
EOF

cp kubernetes.conf /etc/sysctl.d/kubernetes.conf
modprobe ip_conntrack
sysctl -p /etc/sysctl.d/kubernetes.conf

# Log retention
mkdir /etc/systemd/journal.conf.d
cat >/etc/systemd/journal.conf.d/99-prophet.conf <<EOF
[Journal]
# Persist logs to disk
Storage=persistent
# Compress historical logs
Compress=yes
SyncIntervalSec=5m
RateLimitInterval=30s
RateLimitBurst=1000
# Max disk usage 10G
SystemMaxUse=10G
# Max size per log file 200M
SystemMaxFileSize=200M
# Keep logs for 2 weeks
MaxRetentionSec=2week
# Do not forward logs to syslog
ForwardToSyslog=no
EOF

systemctl restart systemd-journald

# Prerequisites for enabling IPVS mode in kube-proxy
yum install -y ipset ipvsadm
modprobe br_netfilter
cat >/etc/sysconfig/modules/ipvs.modules <<EOF
#!/bin/bash
modprobe -- ip_vs
modprobe -- ip_vs_rr
modprobe -- ip_vs_wrr
modprobe -- ip_vs_sh
modprobe -- nf_conntrack_ipv4
EOF
chmod 755 /etc/sysconfig/modules/ipvs.modules
bash /etc/sysconfig/modules/ipvs.modules && lsmod |grep -e ip_vs -e nf_conntrack_ipv4

wget -O /etc/yum.repos.d/docker-ce.repo https://mirrors.aliyun.com/docker-ce/linux/centos/docker-ce.repo
yum install -y containerd.io

cat << EOF > /etc/modules-load.d/containerd.conf
overlay
br_netfilter
EOF

modprobe overlay
modprobe br_netfilter

mkdir -p /etc/containerd
containerd config default > /etc/containerd/config.toml

vim /etc/containerd/config.toml
125 SystemdCgroup = true
61 sandbox_image = "/google_containers/pause:3.7"

systemctl restart containerd
systemctl enable containerd

cat > /etc/yum.repos.d/kubernetes.repo << EOF
[kubernetes]
name=Kubernetes
baseurl=https://mirrors.aliyun.com/kubernetes/yum/repos/kubernetes-el7-x86_64
enabled=1
gpgcheck=0
repo_gpgcheck=0
gpgkey=https://mirrors.aliyun.com/kubernetes/yum/doc/yum-key.gpg https://mirrors.aliyun.com/kubernetes/yum/doc/rpm-package-key.gpg
EOF

yum makecache
yum install  kubectl kubeadm kubelet -y

cat > /etc/sysconfig/kubelet <<EOF
KUBELET_EXTRA_ARGS="--cgroup-driver=systemd"
EOF
systemctl enable kubelet

Deploy the master node

kubeadm init \
--apiserver-advertise-address=192.168.209.128 \
--image-repository registry.aliyuncs.com/google_containers \
--kubernetes-version v1.25.0 \
--service-cidr=10.1.0.0/16 \
--pod-network-cidr=10.244.0.0/16 --v=6

## If it fails, run kubeadm reset -f
journalctl -u kubelet

mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config
export KUBECONFIG=/etc/kubernetes/admin.conf

kubeadm join 192.168.209.128:6443 --token 2k9emz.egvunxvhti1avfs3 \
    --discovery-token-ca-cert-hash sha256:18d09ae702b558269aa4f0fe901b0cac1a9b78e0dd9d9e5c1b3f3899632a2cd7

kubectl get nodes
kubectl get pods -n kube-system

Deploy

wget https://docs.projectcalico.org/manifests/calico.yaml --no-check-certificate

vim +4546 calico.yaml
...
- name: CALICO_IPV4POOL_CIDR
  value: "10.244.0.0/16"

kubectl apply -f calico.yaml
```

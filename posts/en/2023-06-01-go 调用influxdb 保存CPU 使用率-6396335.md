> **About this post**
>
> This post uses gopsutil to collect CPU usage per core (with a 1-second
> sampling interval), then organizes each core's usage into
> cpu data points and writes them one by one to InfluxDB via
> influxdb-client-go/v2. The complete example code is included at the end.

> **Technical notes**
>
> influxdb-client-go/v2 uses the org/bucket model:
> the two arguments of WriteAPIBlocking("test", "test") are the organization
> and the bucket, respectively, unlike the database/retention policy
> style of the v1 SDK;
> data points are converted to Line Protocol by the SDK before being written.
> cpu.Percent(interval, true) first blocks for one sampling interval,
> then returns the usage of each core; cpu.Counts(true) returns the number
> of logical cores. NewClient("http://", "1") is a redacted placeholder and
> should be replaced with the actual
> server URL and API token; in the original, "id": id referenced an undefined
> variable — it has been changed to machineid to match the parameter, and the
> github.com/ prefix that was stripped from the imports has been restored.

## 1. Complete Example Code

```go
package main

import (
    "context"
    "fmt"
    "github.com/influxdata/influxdb-client-go/v2/api/write"
    "github.com/shirou/gopsutil/cpu"
    "log"
    "time"

    influxdb2 "github.com/influxdata/influxdb-client-go/v2"
)

func main(machineid string) {
    // Connect to InfluxDB
    client := influxdb2.NewClient("http://", "1")

    // Create the write API
    writeAPI := client.WriteAPIBlocking("test", "test")

    cpuCount, err := cpu.Counts(true)
    if err != nil {
        log.Fatal("Failed to get CPU core count:", err)
    }
    // Set the sampling interval
    interval := time.Second

    // Collect CPU usage for each core
    cpuPercent, err := cpu.Percent(interval, true)
    if err != nil {
        log.Println("Failed to get CPU usage:", err)
    }

    // Create the data point slice
    points := make([]*write.Point, cpuCount)
    for i := 0; i < cpuCount; i++ {
        fmt.Println(i, cpuPercent[i])
        // Create a data point
        points[i] = influxdb2.NewPoint("cpu",
            map[string]string{"tag1": "id", "tag2": "cpu", "tag3": "percent"},
            map[string]interface{}{"id": machineid, "cpu": i, "percent": cpuPercent[i]},
            time.Now())
        // Write the data point
        err = writeAPI.WritePoint(context.Background(), points[i])
        if err != nil {
            log.Println("Failed to write data points:", err)
        }
    }

    // Close the connection
    client.Close()
}
```

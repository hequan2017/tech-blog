> **About this post**
>
> This post provides a complete nginx.conf for Nginx + Tomcat dynamic/static
> separation: static files (html/js/css/images, etc.) are served directly by
> Nginx from the Tomcat webapps/ROOT directory with expires caching of 1d to
> 30d; jsp/jspx/do requests are forwarded via proxy_pass to the backend
> 192.168.10.51:8080, passing along the X-Real-IP / X-Forwarded-For headers
> and configuring proxy buffering and timeouts. The global section also
> includes epoll, gzip compression, connection limits, version hiding
> (server_tokens off), denial of hidden directories other than .well-known
> and of WEB-INF access, plus commented-out PHP/status blocks for reference.

> **Technical notes**
>
> 1. CentOS 7 reached end of life (EOL) on June 30, 2024; consider migrating
>    to Rocky Linux 9 / AlmaLinux 9 or the domestically developed openEuler.
> 2. In the JSP-forwarding regex `~ \.(jsp|jspx|do)?$`, the `?` makes the
>    extension optional, so requests "ending with a dot" are matched as well;
>    for new deployments it is recommended to write `~ \.(jsp|jspx|do)$`.
> 3. This architecture (front-end Nginx static caching + reverse proxying to
>    Tomcat) is still the mainstream approach today; in production it is
>    recommended to also define an upstream block for the backend pool and
>    add health checks.

## 1. Notes

The installation steps are omitted. Below is the nginx configuration file. Remember to change the directory permissions to the nginx user.

## 2. Complete nginx.conf Configuration

```nginx
user  nginx nginx;
worker_processes auto;
error_log    /var/log/nginx/error.log      crit;
pid           /var/run/nginx.pid;
worker_rlimit_nofile 51200;
events
    {
        use epoll;
        worker_connections 51200;
        multi_accept on;
    }
http
    {

        include       mime.types;
        default_type  application/octet-stream;

        log_format  main  '$remote_addr - $remote_user [$time_local] "$request" '
                          '$status $body_bytes_sent "$http_referer" '
                          '"$http_user_agent" "$http_x_forwarded_for"';


        server_names_hash_bucket_size 128;
        client_header_buffer_size 32k;
        large_client_header_buffers 4 32k;
        client_max_body_size 50m;
        sendfile   on;
        tcp_nopush on;
        keepalive_timeout 60;
        tcp_nodelay on;
        fastcgi_connect_timeout 300;
        fastcgi_send_timeout 300;
        fastcgi_read_timeout 300;
        fastcgi_buffer_size 64k;
        fastcgi_buffers 4 64k;
        fastcgi_busy_buffers_size 128k;
        fastcgi_temp_file_write_size 256k;


        gzip on;
        gzip_min_length  1k;
        gzip_buffers     4 16k;
        gzip_http_version 1.1;
        gzip_comp_level 2;
        gzip_types     text/plain application/javascript application/x-javascript text/javascript text/css application/xml application/xml+rss;
        gzip_vary on;
        gzip_proxied   expired no-cache no-store private auth;
        gzip_disable   "MSIE [1-6]\.";

        #limit_conn_zone $binary_remote_addr zone=perip:10m;  uses 10M of shared memory for per-IP connection overhead
        ##If enable limit_conn_zone,add "limit_conn perip 10;" to server section. 10 connections per IP; add it inside the location block


        server_tokens off;
        access_log off;

server
    {
        listen 80 default_server;
        #listen [::]:80 default_server ipv6only=on;
        server_name  localhost;


        location / {
            root /usr/local/tomcat/webapps/ROOT;
            index  index.htm index.html index.jsp;
        }



        location ~ \.(jsp|jspx|do)?$ {
            index index.jsp;
            proxy_pass http://192.168.10.51:8080;  #jsp requests are handed to tomcat
            proxy_redirect off;
            proxy_set_header Host $host;  #the backend web server can get the client's real IP via X-Forwarded-For
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            client_max_body_size 10m;  #maximum single-file size (bytes) allowed in client requests
            client_body_buffer_size 128k; #maximum bytes buffered by the proxy for client requests
            proxy_connect_timeout 90;  #nginx-to-backend connection timeout (proxy connect timeout)
            proxy_read_timeout 90;   #backend response time after a successful connection (proxy read timeout)
            proxy_buffer_size 4k;    #buffer size for the proxy (nginx) to hold client header information
            proxy_buffers 6 32k;    #proxy_buffers; use this setting if pages average under 32k
            proxy_busy_buffers_size 64k;#buffer size under heavy load (proxy_buffers*2)
            proxy_temp_file_write_size 64k; #temp cache folder size; if larger than this, data is streamed from the upstream server
        }


        location ~ .*\.(html|htm|gif|jpg|jpeg|bmp|png|ico|txt|js|css)$ {
            root  /usr/local/tomcat/webapps/ROOT;
            expires 1d;
        }


        location ~ ^/(WEB-INF)/ {
            deny all;
        }


        error_page   404   /404.html;


        ##PHP


        # Deny access to PHP files in specific directory
        #location ~ /(wp-content|uploads|wp-includes|images)/.*\.php$ { deny all; }

        # location ~ \.php$ {

        #     fastcgi_param  SCRIPT_FILENAME      $document_root$fastcgi_script_name;
        #    include        fastcgi_params;
        #  }


        #location ~ [^/]\.php(/|$)
        #{
        #   try_files $uri =404;
        #   fastcgi_pass  unix:/tmp/php-cgi.sock;
        #   fastcgi_index index.php;
        #   include fastcgi.conf;
        #}
        # location /nginx_status
        # {
        #    stub_status on;
        #    access_log   off;
        # }


        location ~ .*\.(gif|jpg|jpeg|png|bmp|swf)$
        {
            expires      30d;
        }


        location ~ .*\.(js|css)?$
        {
            expires      12h;
        }


        location ~ /.well-known {
            allow all;
        }



        location ~ /\.
        {
            deny all;
```

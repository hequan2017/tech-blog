> **About this post**
>
> This post uses Go to read an `ip` file and parse addresses line by line, sends ICMP ping probes with the `go-fastping` library, and prints an `ok/no` status for each host depending on whether a reply is received. It is suitable as a simple liveness-check script.

> **Technical notes**
>
> The example is written against `tatsushid/go-fastping`; the import path in the original was truncated and should actually be `github.com/tatsushid/go-fastping`. The library has not been updated for years; on Go 1.20+ consider migrating to `github.com/prometheus-community/pro-bing` or wrapping the system `ping` command.

---

```go
package main

import (
	"fmt"
	"io/ioutil"
	"net"
	"os"
	"strings"
	"time"

	"github.com/tatsushid/go-fastping"
)

func main() {
	var result []string
	b, err := ioutil.ReadFile("ip")
	if err != nil {
	}
	s := string(b)
	for _, lineStr := range strings.Split(s, "\n") {
		lineStr = strings.TrimSpace(lineStr)
		if lineStr == "" {
			continue
		}
		result = append(result, lineStr)
	}
	for _, v := range result {
		p := fastping.NewPinger()
		ra, err := net.ResolveIPAddr("ip4:icmp", v)
		if err != nil {
			fmt.Println(err)
			os.Exit(1)
		}
		p.AddIPAddr(ra)
		var d int
		p.OnRecv = func(addr *net.IPAddr, rtt time.Duration) {
			d = 1
		}
		p.OnIdle = func() {
			if d == 1 {
				fmt.Println(v, "ok")
			} else {
				fmt.Println(v, "no")
			}
		}
		err = p.Run()
		if err != nil {
			fmt.Println(err)
		}
	}
}
```

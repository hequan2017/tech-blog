> **About this post**
>
> This post implements a download rate limiter with the token-bucket limiter from `golang.org/x/time/rate`:
> a custom `rateLimitedReader` wraps the HTTP response body and calls `WaitN` after each `Read`
> based on the number of bytes actually read; on the download side, `io.Copy + io.TeeReader`
> copies and counts at the same time, with the sample rate limited to 100 KB/s.

> **Technical notes**
>
> The idea is simply a token bucket: `rate.Limit(100*1024)` counts "events per second" in bytes,
> and the bucket capacity is also 100 KB; `WaitN(ctx, n)` blocks until n tokens have accumulated,
> turning the bytes read into a wait on traffic. `io.TeeReader(limitedReader, counter)` lets `io.Copy`
> hand the data to `counter` for counting while downloading, while the destination `io.Discard`
> merely discards it. The original code returns as soon as it reads EOF, so the last chunk of data
> goes through without waiting — this is inherent to the snippet form; `req`, `counter`, `downloadDone`,
> and `log` are variables from the surrounding code, which the original post does not define either.

```go
import (
	"context"
	"errors"
	"io"

	"golang.org/x/time/rate"
)

type rateLimitedReader struct {
	r       io.Reader
	limiter *rate.Limiter
}

func (rlr *rateLimitedReader) Read(p []byte) (int, error) {
	n, err := rlr.r.Read(p)
	if err != nil {
		return n, err
	}
	if n > 0 {
		// WaitN blocks until the limiter allows n events to happen
		err = rlr.limiter.WaitN(context.Background(), n)
		if err != nil {
			return 0, err
		}
	}
	return n, nil
}

// Create a limiter with a rate of 100 KB/s
limiter := rate.NewLimiter(rate.Limit(100*1024), int(100*1024)) // 100 KB per second

doDownload := func() {
	resp, err := client.Do(req)
	if err != nil {
		log.Debugf("Failed when making HTTP request: %s", err)
	} else {
		defer resp.Body.Close()

		// Wrap resp.Body to limit the download speed
		limitedReader := &rateLimitedReader{
			r:       resp.Body,
			limiter: limiter,
		}

		if _, err = io.Copy(io.Discard, io.TeeReader(limitedReader, counter)); err != nil {
			if !errors.Is(err, context.Canceled) && !errors.Is(err, context.DeadlineExceeded) {
				log.Debugf("Failed when reading HTTP response: %s", err)
			}
		}

		downloadDone <- struct{}{}
	}
}
```

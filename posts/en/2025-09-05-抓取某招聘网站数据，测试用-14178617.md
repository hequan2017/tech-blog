> **About this post**
>
> This post records a Python test script for scraping job listings from 58.com:
> it defines an Enhanced58JobScraper class that uses Selenium to drive Chrome through
> the job listing pages of 7 cities including Beijing and Shanghai, pausing for manual
> CAPTCHA handling whenever one appears and then continuing;
> it then visits each job detail page and company page in turn, extracting 20+ fields
> such as job title, salary, education, and contact information with CSS selectors first
> and regex as a fallback. Every job is appended to an Excel file in real time,
> and a JSON backup is saved once everything is done.

> **Technical notes**
>
> - Since Selenium 4.6, Selenium Manager ships built in and can match the ChromeDriver
>   automatically, so the webdriver-manager fallback in this post is no longer necessary;
>   the hardcoded Chrome 91 User-Agent is also clearly outdated — treat this post as a
>   historical example only.
> - In bs4's find_all(text=...), the text parameter is a legacy alias of string=;
>   new code should use string instead.
> - Compliance reminder: before scraping, check the target site's robots.txt and terms of
>   service, and keep the request rate under control. Phone numbers, email addresses,
>   unified social credit codes and the like on the pages are personal information
>   governed by the Personal Information Protection Law (《个人信息保护法》). They are
>   suitable only for local testing and must not be used commercially or distributed.

```python
class Enhanced58JobScraper:
    def __init__(self, headless=True):
        self.options = Options()
        if headless:
            self.options.add_argument("--headless")
        self.options.add_argument("--no-sandbox")
        self.options.add_argument("--disable-dev-shm-usage")
        self.options.add_argument("--disable-blink-features=AutomationControlled")
        self.options.add_experimental_option("excludeSwitches", ["enable-automation"])
        self.options.add_experimental_option('useAutomationExtension', False)
        self.options.add_argument("--user-agent=Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36")

        # Try the ChromeDriver from the system PATH first; fall back to webdriver-manager
        try:
            self.driver = webdriver.Chrome(options=self.options)
        except Exception as e:
            print(f"使用系统ChromeDriver失败，尝试使用webdriver-manager: {e}")
            try:
                service = Service(ChromeDriverManager().install())
                self.driver = webdriver.Chrome(service=service, options=self.options)
            except Exception as e2:
                print(f"webdriver-manager也失败了: {e2}")
                raise e2
        self.driver.execute_script("Object.defineProperty(navigator, 'webdriver', {get: () => undefined})")
```

```python
    def generate_page_urls(self, base_url, max_pages=5):
        """Generate the URL list for multiple pages"""
        url_list = []

        print(f"\n=== 开始生成 {max_pages} 页的URL列表 ===")

        for page in range(1, max_pages + 1):
            if page == 1:
                current_url = base_url
            else:
                # Build URLs following 58.com's actual pagination format
                # Page 1 is the original URL (no pn), page 2 is pn2, page 3 is pn3, and so on
                if "hulianwangtx" in base_url:
                    # Split off the base URL part and the original params
                    if "?" in base_url:
                        base_part = base_url.split('?')[0]  # https://bj.58.com/hulianwangtx/
                        original_params = base_url.split('?')[1]  # keep the original params
                        # Fix: page 2 is pn2 and page 3 is pn3, so just use page directly
                        current_url = f"{base_part}pn{page}/?{original_params}"
                    else:
                        # No params: use the defaults
                        base_part = base_url.rstrip('/')
                        current_url = f"{base_part}/pn{page}/"
                else:
                    # Generic pagination format
                    if "?" in base_url:
                        current_url = f"{base_url}&pagenum={page}"
                    else:
                        current_url = f"{base_url}?pagenum={page}"

            url_list.append(current_url)
            print(f"第 {page} 页URL: {current_url}")

        print(f"\n=== URL列表生成完成，共 {len(url_list)} 个URL ===")
        return url_list

    def scrape_multiple_pages(self, base_url, max_pages=5):
        """Scrape job data from multiple pages - generate the URL list first, then scrape in batch"""
        # Step 1: generate the URL list for all pages
        url_list = self.generate_page_urls(base_url, max_pages)

        # Step 2: scrape the data of all URLs in batch
        all_jobs_data = []

        print(f"\n=== 开始批量抓取 {len(url_list)} 页的职位数据 ===")

        for i, url in enumerate(url_list, 1):
            try:
                print(f"\n--- 正在抓取第 {i} 页 ---")
                print(f"URL: {url}")

                # Scrape the current page's data
                page_data = self.get_job_list_from_page(url)

                if page_data:
                    all_jobs_data.extend(page_data)
                    print(f"第 {i} 页成功抓取到 {len(page_data)} 个职位")
                    print(f"当前总计抓取职位数: {len(all_jobs_data)}")
                else:
                    print(f"第 {i} 页没有抓取到数据，可能已到最后一页")
                    if i > 1:  # not the first page and no data, probably the last page
                        print(f"检测到第 {i} 页无数据，但继续抓取剩余页面")

                # Delay between pages to avoid hitting the site too often
                if i < len(url_list):
                    print(f"等待 3 秒后继续下一页...")
                    time.sleep(3)

            except Exception as e:
                print(f"抓取第 {i} 页时出错: {e}")
                continue

        print(f"\n=== 多页抓取完成，总共抓取到 {len(all_jobs_data)} 个职位 ===")
        return all_jobs_data
```

## 4. Job Link Extraction

```python
    def get_job_links(self):
        """Get the detail links of all jobs on the current page"""
        job_links = []
        try:
            # Try several ways to get job links
            link_selectors = [
                "span.name",  # the span.name element specified by the user
                "a[href*='shtml']",  # 58.com job detail pages usually end with .shtml
                "a[href*='job']",
                "a[href*='zhaopin']",
                ".job_name a",
                ".job-title a",
                ".title a"
            ]

            for selector in link_selectors:
                elements = self.driver.find_elements(By.CSS_SELECTOR, selector)
                for element in elements:
                    if selector == "span.name":
                        # For span.name elements, locate the parent or a related link
                        try:
                            # Try to find the link element containing this span
                            parent_link = element.find_element(By.XPATH, "./ancestor::a[1]")
                            href = parent_link.get_attribute('href')
                        except:
                            # If the span is not inside a link, try a sibling or adjacent link
                            try:
                                sibling_link = element.find_element(By.XPATH, "./following-sibling::a[1] | ./preceding-sibling::a[1]")
                                href = sibling_link.get_attribute('href')
                            except:
                                # If all else fails, try clicking the span itself to see if it is clickable
                                try:
                                    element.click()
                                    time.sleep(2)
                                    current_url = self.driver.current_url
                                    if current_url != self.driver.current_url:
                                        href = current_url
                                    else:
                                        href = None
                                except:
                                    href = None
                    else:
                        href = element.get_attribute('href')

                    if href and ('shtml' in href or 'job' in href or 'zhaopin' in href or 'detail' in href):
                        if href not in job_links:
                            job_links.append(href)

                if job_links:  # stop trying the other selectors once links are found
                    break

        except Exception as e:
            print(f"获取职位链接失败: {e}")

        return job_links
```

```python
    def scrape_job_detail_page(self, job_url):
        """Enter a job detail page and scrape the full information"""
        job_data = {
            "企业名称": "",
            "企业类型": "",
            "社会信用码": "",
            "企业规模": "",
            "注册资本(万)": "",
            "所属区域": "",
            "联系人": "",
            "联系方式": "",
            "联系邮箱": "",
            "办公地址": "",
            "企业简介": "",
            "营业执照": "",
            "企业相册": "",
            "岗位名称": "",
            "薪资类型": "",
            "薪资范围起": "",
            "薪资范围至": "",
            "工作地点": "",
            "岗位要求": "",
            "学历要求": "",
            "招聘人数": "",
            "发布时间": "",
            "结束时间": "",
            "工作职责": "",
            "任职要求": ""
        }

        try:
            # Visit the job detail page
            self.driver.get(job_url)
            time.sleep(1)

            # Wait for the page to load
            WebDriverWait(self.driver, 10).until(
                EC.presence_of_element_located((By.TAG_NAME, "body"))
            )

            # Detect the CAPTCHA and wait for manual handling
            page_source = self.driver.page_source
            if "访问过于频繁，本次访问做以下验证码校验" in page_source or "验证码校验" in page_source:
                print("\n检测到验证码页面，请手动完成验证码验证...")
                print("验证完成后，请按回车键继续...")
                input()  # Wait for the user to press Enter
                print("继续执行...")
                # Re-fetch the page source
                page_source = self.driver.page_source

            # Get the page source
            soup = BeautifulSoup(page_source, "html.parser")

            # Remove recommendation blocks such as "jobs you may like"
            for unwanted in soup.find_all(['div', 'section'], class_=lambda x: x and ('recommend' in x.lower() or 'interest' in x.lower() or 'similar' in x.lower())):
                unwanted.decompose()

            # Remove blocks containing the "you may like" text
            for element in soup.find_all(text=lambda text: text and '您可能感兴趣' in text):
                parent = element.parent
                while parent and parent.name != 'body':
                    if parent.name in ['div', 'section', 'aside']:
                        parent.decompose()
                        break
                    parent = parent.parent

            # Keep only the main content area
            main_content = soup.find('div', class_=lambda x: x and ('main' in x.lower() or 'content' in x.lower() or 'detail' in x.lower()))
            if main_content:
                page_text = main_content.get_text()
            else:
                page_text = soup.get_text()

            # Extract the job title - selectors based on the real HTML structure
            title_selectors = [
                ".pos_title", ".job-title", ".job_title", ".title", "h1", ".name",
                "[class*='title']:not([class*='recommend']):not([class*='similar'])",
                "[class*='name']:not([class*='recommend']):not([class*='similar'])"
            ]
            for selector in title_selectors:
                title_element = soup.select_one(selector)
                if title_element and title_element.get_text().strip():
                    title_text = title_element.get_text().strip()
                    # Filter out texts with recommendation content
                    if '您可能感兴趣' not in title_text and '推荐职位' not in title_text:
                        job_data["岗位名称"] = title_text
                        break

            # Extract the company name and link - selectors based on the real HTML structure
            company_url = None
            company_selectors = [
                ".baseInfo_link a", ".baseInfo_link", ".company-name", ".company_name", ".company", ".corp-name",
                "h1 + .company", ".job-company", ".employer-name",
                "[class*='company']:not([class*='recommend']):not([class*='similar'])"
            ]
            for selector in company_selectors:
                company_element = soup.select_one(selector)
                if company_element and company_element.get_text().strip():
                    company_text = company_element.get_text().strip()
                    # Filter out texts with recommendation content
                    if '您可能感兴趣' not in company_text and '推荐职位' not in company_text:
                        job_data["企业名称"] = company_text
                        # Extract the company link
                        if company_element.name == 'a' and company_element.get('href'):
                            company_url = company_element.get('href')
                            if not company_url.startswith('http'):
                                company_url = 'https:' + company_url if company_url.startswith('//') else 'https://58.com' + company_url
                        break

            # If no company name was found, extract one from the page text with regex
            if not job_data["企业名称"]:
                # Extract from the first half of the page text to avoid recommendation blocks
                text_lines = page_text.split('\n')[:50]  # take only the first 50 lines
                first_half_text = '\n'.join(text_lines)

                company_patterns = [
                    r'([\u4e00-\u9fa5]{2,20}(?:公司|企业|集团|科技|网络|信息|技术|有限|股份))',
                    r'([A-Za-z\s]{2,30}(?:公司|企业|集团|科技|网络|信息|技术|有限|股份|Company|Corp|Inc|Ltd))'
                ]
                for pattern in company_patterns:
                    match = re.search(pattern, first_half_text)
                    if match:
                        job_data["企业名称"] = match.group(1).strip()
                        break

            # Extract the salary info - based on the real HTML structure and text patterns
            # Try CSS selectors first
            salary_element = soup.select_one(".pos_salary")
            if salary_element:
                salary_text = salary_element.get_text().strip()
                salary_match = re.search(r'(\d+)-(\d+)', salary_text)
                if salary_match:
                    job_data["薪资范围起"] = salary_match.group(1)
                    job_data["薪资范围至"] = salary_match.group(2)

            # Fall back to regex if the CSS selector found nothing
            if not job_data["薪资范围起"]:
                salary_patterns = [
                    r'(\d+)[-~](\d+)元/月',
                    r'(\d+)[-~](\d+)万/年',
                    r'(\d+)[-~](\d+)千/月',
                    r'薪资.*?(\d+)[-~](\d+)',
                    r'工资.*?(\d+)[-~](\d+)'
                ]

                # Extract the salary info from the first half of the page text
                text_lines = page_text.split('\n')[:30]  # take only the first 30 lines
                salary_text = '\n'.join(text_lines)

                for pattern in salary_patterns:
                    salary_match = re.search(pattern, salary_text)
                    if salary_match:
                        groups = salary_match.groups()
                        if len(groups) >= 2:
                            job_data["薪资范围起"] = groups[0]
                            job_data["薪资范围至"] = groups[1]
                        break

            # Salary type logic: default to non-negotiable; if no salary range matched, set it to negotiable
            if job_data["薪资范围起"] and job_data["薪资范围至"]:
                job_data["薪资类型"] = "非面谈"
            else:
                job_data["薪资类型"] = "面谈"

            # Extract the work location - based on the real HTML structure, detailed address first
            # First try to extract the detailed-address span from the .pos-area div
            pos_area_div = soup.select_one(".pos-area")
            detailed_address = ""

            if pos_area_div:
                # Look for span tags containing the detailed address (spans without a class)
                address_spans = pos_area_div.find_all("span")
                for span in address_spans:
                    # Skip spans that have a class attribute
                    if not span.get("class"):
                        span_text = span.get_text().strip()
                        # Check whether it contains detailed address info (district or street address)
                        if re.search(r'[\u4e00-\u9fa5]+(?:区|市)[\u4e00-\u9fa5\w\(\)]+|[\u4e00-\u9fa5]+[\u4e00-\u9fa5]+[\u4e00-\u9fa5\w\(\)]+(?:大厦|中心|广场|大楼|科技园|金融谷|商务区|写字楼|园区|基地)', span_text):
                            detailed_address = span_text
                            break

            # Use the detailed address directly if found
            if detailed_address:
                job_data["工作地点"] = detailed_address
            else:
                # Fallback: extract the basic info from .pos_area_item
                location_elements = soup.select(".pos_area_item")
                if location_elements and len(location_elements) >= 2:
                    # Combine city and district info
                    location_parts = [elem.get_text().strip() for elem in location_elements[:3]]
                    basic_location = "-".join(location_parts)
                    job_data["工作地点"] = basic_location

            # Fall back to regex if the CSS selector found nothing
            if not job_data["工作地点"]:
                location_patterns = [
                    r'工作地点.*?([\u4e00-\u9fa5]+市[\u4e00-\u9fa5]+区[\u4e00-\u9fa5\w\s]+)(?=\n|联系|电话|$)',
                    r'地址.*?([\u4e00-\u9fa5]+市[\u4e00-\u9fa5]+区[\u4e00-\u9fa5\w\s]+)(?=\n|联系|电话|$)',
                    r'([\u4e00-\u9fa5]+市[\u4e00-\u9fa5]+区[\u4e00-\u9fa5\w\s]+(?:大厦|中心|广场|大楼|科技园|金融谷|商务区|写字楼|园区|基地))',
                    r'([\u4e00-\u9fa5]+市[\u4e00-\u9fa5]+区[\u4e00-\u9fa5]+.*?)(?=\n|$)',
                    r'(北京市?[\u4e00-\u9fa5]+区[\u4e00-\u9fa5\w\s]+)(?=\n|$)',
                    r'(上海市?[\u4e00-\u9fa5]+区[\u4e00-\u9fa5\w\s]+)(?=\n|$)',
                    r'(广州市?[\u4e00-\u9fa5]+区[\u4e00-\u9fa5\w\s]+)(?=\n|$)',
                    r'(深圳市?[\u4e00-\u9fa5]+区[\u4e00-\u9fa5\w\s]+)(?=\n|$)'
                ]

                # Extract the location info from the first half of the page text
                location_text = '\n'.join(page_text.split('\n')[:40])

                for pattern in location_patterns:
                    location_match = re.search(pattern, location_text)
                    if location_match:
                        job_data["工作地点"] = location_match.group(1).strip()
                        break

            # Extract the education requirement - based on the real HTML structure
            # Try CSS selectors first
            condition_elements = soup.select(".item_condition")
            for elem in condition_elements:
                text = elem.get_text().strip()
                if any(edu in text for edu in ['博士', '硕士', '研究生', '本科', '大专', '专科', '高中', '中专', '学历不限']):
                    job_data["学历要求"] = text
                    break

            # Fall back to regex if the CSS selector found nothing
            if not job_data["学历要求"]:
                education_patterns = [
                    r'学历要求.*?(博士|硕士|研究生|本科|大专|专科|高中|中专|不限)',
                    r'学历.*?(博士|硕士|研究生|本科|大专|专科|高中|中专|不限)',
                    r'(博士|硕士|研究生|本科|大专|专科)以上',
                    r'要求.*?(博士|硕士|研究生|本科|大专|专科|高中|中专)'
                ]

                # Extract the education info from the first half of the page text
                education_text = '\n'.join(page_text.split('\n')[:50])

                for pattern in education_patterns:
                    edu_match = re.search(pattern, education_text)
                    if edu_match:
                        job_data["学历要求"] = edu_match.group(1)
                        break

            # Extract the experience requirement - based on the real HTML structure
            # Try CSS selectors first
            for elem in condition_elements:
                text = elem.get_text().strip()
                if any(exp in text for exp in ['经验', '年', '应届', '不限']):
                    if '学历' not in text and '招' not in text:  # exclude education and headcount info
                        job_data["岗位要求"] = text
                        break

            # Fall back to regex if the CSS selector found nothing
            if not job_data["岗位要求"]:
                exp_patterns = [
                    r'工作经验.*?(\d+)[-~](\d+)年',
                    r'经验.*?(\d+)[-~](\d+)年',
                    r'(\d+)年以上.*?经验',
                    r'经验.*?(\d+)年以上',
                    r'(无需经验|不限经验|应届毕业生|经验不限)'
                ]

                # Extract the experience info from the first half of the page text
                exp_text = '\n'.join(page_text.split('\n')[:50])

                for pattern in exp_patterns:
                    exp_match = re.search(pattern, exp_text)
                    if exp_match:
                        if len(exp_match.groups()) >= 2:
                            job_data["岗位要求"] = f"{exp_match.group(1)}-{exp_match.group(2)}年经验"
                        else:
                            job_data["岗位要求"] = exp_match.group(1)
                        break

            # Extract the headcount - based on the real HTML structure
            # Try CSS selectors first
            for elem in condition_elements:
                text = elem.get_text().strip()
                if '招' in text and '人' in text:
                    job_data["招聘人数"] = text
                    break

            # Fall back to regex if the CSS selector found nothing
            if not job_data["招聘人数"]:
                recruit_patterns = [
                    r'招聘.*?(\d+)人',
                    r'招.*?(\d+)人',
                    r'(\d+)人',
                    r'(若干)人?',
```

```python
                            elif "年" in salary_match.group(0):
                                job_data["薪资类型"] = "年薪"
                            else:
                                job_data["薪资类型"] = "月薪"
                    break

            # Extract the work location
            location_patterns = [
                r'(北京市?[\u4e00-\u9fa5]+区)',
                r'(上海市?[\u4e00-\u9fa5]+区)',
                r'(广州市?[\u4e00-\u9fa5]+区)',
                r'(深圳市?[\u4e00-\u9fa5]+区)',
                r'([\u4e00-\u9fa5]+市[\u4e00-\u9fa5]+区)',
                r'([\u4e00-\u9fa5]+区)'
            ]

            for pattern in location_patterns:
                location_match = re.search(pattern, item_text)
                if location_match:
                    job_data["工作地点"] = location_match.group(1)
                    break

            # Extract the education requirement
            education_patterns = [
                r'(博士|硕士|研究生|本科|大专|专科|高中|中专|不限).*学历',
                r'学历.*(博士|硕士|研究生|本科|大专|专科|高中|中专|不限)',
                r'(博士|硕士|研究生|本科|大专|专科)以上',
                r'(博士|硕士|研究生|本科|大专|专科|高中|中专)'
            ]

            for pattern in education_patterns:
                edu_match = re.search(pattern, item_text)
                if edu_match:
                    job_data["学历要求"] = edu_match.group(0)
                    break

            # Extract the work experience
            exp_patterns = [
                r'(\d+)[-~](\d+)年.*经验',
                r'经验.*(\d+)[-~](\d+)年',
                r'(\d+)年以上.*经验',
                r'经验.*(\d+)年以上',
                r'(无需经验|不限经验|应届毕业生)'
            ]

            for pattern in exp_patterns:
                exp_match = re.search(pattern, item_text)
                if exp_match:
                    job_data["岗位要求"] = exp_match.group(0)
                    break

            # Extract the publish time
            time_patterns = [
                r'(\d{4}-\d{2}-\d{2})',
                r'(\d{2}-\d{2})',
                r'(今天|昨天|前天)',
                r'(\d+)小时前',
                r'(\d+)天前'
            ]

            for pattern in time_patterns:
                time_match = re.search(pattern, item_text)
                if time_match:
                    job_data["发布时间"] = time_match.group(0)
                    break

            # Try to extract the company name (usually near the job)
            company_patterns = [
                r'([\u4e00-\u9fa5]+(?:公司|企业|集团|科技|网络|信息|技术|有限|股份))',
                r'([A-Za-z]+(?:公司|企业|集团|科技|网络|信息|技术|有限|股份))'
            ]

            for pattern in company_patterns:
                company_match = re.search(pattern, item_text)
                if company_match:
                    job_data["企业名称"] = company_match.group(1)
                    break

        except Exception as e:
            print(f"提取职位项目信息失败: {e}")

        return job_data
```

```python
def main():

        ]

                    page_data = scraper.get_job_list_from_page(url)

                    if page_data:
                        city_data.extend(page_data)
                        print(f"    第{page_num}页抓取到 {len(page_data)} 个职位")
                    else:
                        print(f"    第{page_num}页未抓取到职位数据")

                    # Delay between pages
                    if page_num < len(url_list):
                        time.sleep(3)

                if city_data:
                    # Tag each job with the city
                    for job in city_data:
                        job['抓取城市'] = city_name

                    all_data.extend(city_data)
                    print(f"{city_name}成功抓取到 {len(city_data)} 个职位信息")
                else:
                    print(f"{city_name}未抓取到任何职位数据")

            except Exception as e:
                print(f"{city_name}抓取出错: {e}")
                continue  # move on to the next city

        if all_data:
            print(f"\n所有城市总共成功抓取到 {len(all_data)} 个职位信息")

            scraper.save_to_json(all_data, "58同城多城市职位详细信息.json")

        else:
            print("所有城市都未抓取到任何职位数据")

    except Exception as e:
        print(f"程序执行出错: {e}")
    finally:
        scraper.close()

if __name__ == "__main__":
    main()
```

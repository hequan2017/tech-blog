> **About this post**
>
> Demonstrates the common Redis APIs via a redis-py connection pool (strings, sets, hashes, lists), and shows a simple List-based message queue example: the producer writes JSON tasks with `rpush`, and the consumer fetches tasks in batches with `lpop` and processes them concurrently via `multiprocessing.Pool`.

> **Technical notes**
>
> The examples are written against redis-py 2.x / Python 3; since redis-py 3.x, return values are bytes, so you need `decode_responses=True` or manual decoding. A few methods such as `substr` and `mset` have been replaced by mappings like `getrange`/`mset`, or had their behavior adjusted, in redis-py 3+; treat the official documentation as authoritative.

### Redis operations

```python
import redis
redisPool = redis.ConnectionPool(host='192.168.100.50', port=6379, db=8)
redis= redis.Redis(connection_pool=redisPool)

redis.set('key','values')
redis.get('com')
redis.append('keys','values')
redis.delete('keys')

print(redis.getset('name','Mike'))  #Set name to Mike and return the previous value
print(redis.mget(['name','age']))   #Print the values of the name and age keys
print(redis.setnx('newname','james'))  #Set the value only if the key does not exist
print(redis.mset({'name1':'smith','name2':'curry'}))  #Batch assignment
print(redis.msetnx({'name3':'ltf','name4':'lsq'}))    #Batch assignment only if the keys do not exist
print(redis.incr('age',1))   #Increment the value of age by 1
print(redis.decr('age',5))   #Decrement the value of age by 5
print(redis.append('name4','is a sb'))   #Append "is a sb" to the value of name4 and return the string length
print(redis.substr('name',1,4))   #Substring of the key name

print(redis.sadd('tags','Book','Tea','Coffee'))  #Returns the set length: 3
print(redis.srem('tags','Book'))  #Returns the number of deleted items
print(redis.spop('tags'))   #Removes and returns a random element
print(redis.smove('tags','tags1','Coffee'))
print(redis.scard('tags'))  # Get the number of elements in the tags set
print(redis.sismember('tags', 'Book'))  # Check whether Book is a member of the tags set
print(redis.sinter('tags', 'tags1'))  # Return the intersection of sets tags and tags1
print(redis.sunion('tags', 'tags1'))  # Return the union of sets tags and tags1
print(redis.sdiff('tags', 'tags1'))  # Return the difference of sets tags and tags1
print(redis.smembers('tags'))  # Return all elements of the tags set

print(redis.hset('price','cake',5))  # Add a mapping to the hash named price; returns 1, the number of mappings added
print(redis.hsetnx('price','book',6)) # Add a mapping to the hash named price; returns 1, the number of mappings added
print(redis.hget('price', 'cake'))  # Get the value of cake; returns 5
print(redis.hmset('price',{'banana':2,'apple':3,'pear':6,'orange':7}))   #Add mappings in batch
print(redis.hmget('price', ['apple', 'orange']))  # Query the values of apple and orange; prints b'3',b'7'
print(redis.hincrby('price','apple',3))   #Increment the apple mapping by 3, to 6
print(redis.hexists('price', 'banana'))  # Whether banana exists in price; returns True
print(redis.hdel('price','banana'))    #Delete banana from price; returns 1
print(redis.hlen('price'))  # Print the length of price
print(redis.hkeys('price'))  # Print all mapping keys
print(redis.hvals('price'))  # Print all mapping values
print(redis.hgetall('price'))  # Print all mapping pairs

print(redis.rpush('list',1,2,3)) #Append 1, 2, 3 to the tail of the list named list; returns the length
print(redis.lpush('list',0))   #Prepend 0 to the head of the list named list; returns the length
print(redis.llen('list'))   #Return the length of the list
print(redis.lrange('list',1,3))  #Return the list slice for the index range from 1 to 3
print(redis.lindex('list',1))   #Return the element at index 1 — value
print(redis.lset('list',1,5))  #Reassign the element at index 1 of the list to 5
print(redis.lpop('list'))  #Remove the first element of list
print(redis.rpop('list'))   #Remove the last element of list
print(redis.blpop('list'))   #Remove the first element of list
print(redis.brpop('list'))    #Remove the last element
print(redis.rpoplpush('list','list1'))   #Remove the tail element of list and push it onto the head of list1
```

### Message queue example

```python
import redis
import json
redisPool = redis.ConnectionPool(host='192.168.100.50', port=6379, db=8)
client = redis.Redis(connection_pool=redisPool)

# Insert data into the redis queue sequentially; the sort parameter is used to verify the pop order
while True:
    num = 0
    for i in range(0, 100):
        num = num + 1
        # params info
        params_dict = {"name": f"test {num}", "sort":num}

        client.rpush("test", json.dumps(params_dict))

    # Inspect the target queue data
    result = client.lrange("test", 0, 100)
    print(result)
    import time
    time.sleep(10)
```

```python
import redis
import time
import multiprocessing
import time
import os
import random

redisPool = redis.ConnectionPool(host='192.168.100.50', port=6379, db=8)
client = redis.Redis(connection_pool=redisPool)

def test1(msg):
    t_start = time.time()
    print("%s开始执行，进程号为%d" % (msg, os.getpid()))
    time.sleep(random.random() * 2)
    t_stop = time.time()
    print("%s执行完成，耗时%.2f" % (msg, t_stop - t_start))

while True:
    number = client.llen('test')
    print("现在的队列任务 条数是 ", number)
    p = 100
    if number > p-1:
        print("-----start-----")
        a = []
        for i in range(p):
            result = client.lpop("test")
            a.append(result)
        print("每10条读取一次", a)
        po = multiprocessing.Pool(p)
        for i in range(0, p):
            # Pool().apply_async(the target to call, (tuple of arguments passed to the target,))
            # Each loop uses an idle child process to call the target
            po.apply_async(test1, (a[i],))
        po.close()  # Close the process pool; po no longer accepts new requests after this
        po.join()  # Wait for all child processes in po to finish; must come after the close statement
        print("-----end-----")
        time.sleep(2)
    elif number < p and number > 0:
        print("-----start-----")
        a = []
        for i in range(number):
            a = []
            result = client.lpop("test")
            a.append(result)
        print("小于10条的 读取一次 ", a)
        po = multiprocessing.Pool(number)
        for i in a:
            # Pool().apply_async(the target to call, (tuple of arguments passed to the target,))
            # Each loop uses an idle child process to call the target
            po.apply_async(test1, (a,))

        po.close()  # Close the process pool; po no longer accepts new requests after this
        po.join()  # Wait for all child processes in po to finish; must come after the close statement
        print("-----end-----")
        time.sleep(2)
    elif number == 0:
        print("没有任务需要处理")
        time.sleep(2)
    else:
        time.sleep(2)
```

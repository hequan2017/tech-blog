> **About this post**
>
> This post is a go-prompt console menu example: the main loop first registers
> Ctrl+C/TERM signal handling, then walks down the completion prompts level by
> level — the main menu offers a choice between "NIC configuration / dial-up
> management", the second-level menu then picks a NIC number or the address
> acquisition mode, next the IP and netmask are entered and printed, and
> finally a goto jumps back to the main menu loop. The whole post is one
> complete piece of example code.

> **Technical notes**
>
> go-prompt (github.com/c-bata/go-prompt), inspired by Python's prompt-toolkit,
> focuses on interactive completion and is positioned differently from cobra
> (subcommand and flag parsing) and readline-style libraries; the latest module
> release is v0.2.6 (published 2021-03), with no releases since. os.Kill
> (SIGKILL) registered via `signal.Notify` cannot be caught on Unix-like
> systems, and SIGTERM is not delivered on Windows either; the only signal
> actually intercepted is Ctrl+C (os.Interrupt), which is why the code comment
> keeps an os.Exit forced-exit fallback. The scraped text had at one point lost
> the `github.com/` prefix of the import and the code block was labeled python;
> both have been fixed.

## 1. Complete Example Code

```go
package main

import (
  "fmt"
  "github.com/c-bata/go-prompt"
  "os"
  "os/signal"
  "syscall"
)

func base(d prompt.Document) []prompt.Suggest {
  s := []prompt.Suggest{
    {Text: "1", Description: "网卡配置"},
    {Text: "2", Description: "拨号管理"},
  }
  return prompt.FilterHasPrefix(s, d.GetWordBeforeCursor(), true)
}

func network1(d prompt.Document) []prompt.Suggest {
  s := []prompt.Suggest{
    {Text: "1", Description: "网卡1"},
    {Text: "2", Description: "网卡2"},
  }
  return prompt.FilterHasPrefix(s, d.GetWordBeforeCursor(), true)
}

func network2(d prompt.Document) []prompt.Suggest {
  s := []prompt.Suggest{
    {Text: "1", Description: "自动获取"},
    {Text: "2", Description: "手动配置"},
  }
  return prompt.FilterHasPrefix(s, d.GetWordBeforeCursor(), true)
}

func input(prompt string) string {
  var text string
  fmt.Print(prompt)
  _, err := fmt.Scan(&text)
  if err != nil {
    return ""
  }
  return text
}

func exitHandle(exitChan chan os.Signal) {
  for {
    select {
    case sig := <-exitChan:
      _ = sig
      //os.Exit(1) // if Ctrl+C cannot stop the program, use os.Exit to force quit
    }
  }
}

func main() {

  exitChan := make(chan os.Signal)
  signal.Notify(exitChan, os.Interrupt, os.Kill, syscall.SIGTERM)
  go exitHandle(exitChan)

LOOP:
  fmt.Println("请选择 \n" +
    "1 网卡配置 \n2 拨号管理 \n0 返回")
  t := prompt.Input("> ", base)

  if t == "1" {

    fmt.Println("可选择 1  网卡1    2 网卡2")
    t1 := prompt.Input("> ", network1)
    fmt.Println("You selected 网卡" + t1)
    fmt.Println("可选择 1  自动获取    2 手动配置")
    t2 := prompt.Input("> ", network2)
    fmt.Println("You selected 网卡" + t2)
    a := input("ip:")
    b := input("network: ")
    fmt.Println(a)
    fmt.Println(b)
  }

  goto LOOP

}
```

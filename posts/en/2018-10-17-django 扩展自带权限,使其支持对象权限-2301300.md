> **About this post**
>
> Building on Django's built-in permissions (users/groups/Permission) without rewriting the original system, this post implements object-level (row-level) permission control through three custom tables (Request for request types, RolePermission for role-group permissions, and Role for group-permission relationships): a JSON field stores filter conditions, and decorators wrap the CBV methods `get_queryset`/`dispatch`/`form_valid` to filter data by those conditions. Suitable for small projects.

> **Technical notes**
>
> The idea still works today, but Django projects now more commonly use django-guardian (for object permissions) or DRF's `DjangoObjectPermissions` to implement object-level permissions, which are more mature than rolling your own. The third-party `jsonfield` library this post depends on is deprecated; the built-in `models.JSONField` in Django 3.1+ is a drop-in replacement.

### Extending Django's Built-in Permissions

#### Description

> Add support for object-level permissions on top of the built-in permission system without rewriting it. Suitable for small projects.
> Comments and suggestions for improvement are welcome.

#### Database

> Create 3 new tables

```python
from django.db import models
from django.contrib.auth.models import AbstractUser, Group, User

from jsonfield import JSONField


class Request(models.Model):
    request = models.CharField(max_length=16, verbose_name='请求类型(大写)')

    class Meta:
        db_table = "request"
        verbose_name = "请求类型"
        verbose_name_plural = verbose_name

    def __str__(self):
        return self.request


class RolePermission(models.Model):
    role = models.CharField(max_length=32, verbose_name='角色组')
    table = models.CharField(max_length=32, verbose_name='表名字')
    request = models.ManyToManyField(Request, verbose_name='请求', related_name='re', )
    permission = JSONField(max_length=1024, verbose_name='权限条件')

    class Meta:
        db_table = "role_permission"
        verbose_name = "角色组权限"
        verbose_name_plural = verbose_name

    def __str__(self):
        return self.role


class Role(models.Model):
    group = models.ForeignKey(Group, verbose_name='用户组', on_delete=models.CASCADE)
    roles = models.ManyToManyField(RolePermission, verbose_name='角色组权限', blank=True, related_name='roles')

    class Meta:
        db_table = "role"
        verbose_name = "角色组关系"
        verbose_name_plural = verbose_name
```

The relationships among the three tables:

```text
Role                 角色组关系    : 系统用户组  <-->  角色组权限
Request              请求类型      : GET ,POST
RolePermission       角色组权限    : 角色  表名字  请求  权限条件(JSON类型)
```

> The RolePermission table is the key one.

#### Example

- Take the common asset table as an example: table name `asset`, field `groups` (grouped into dev and ops).

Steps for dividing permissions:

- Create user hequan
- Create group dev
- In the Request table, add GET (read-only) and POST (update, delete)
- In RolePermission, add:
  - Role: asset-dev read-only
  - Table name: asset (misspelled as assset in the original text)
  - Request: GET
  - Permission condition: `{"groups": "dev"}`
- In the Role table, add: system group dev, role-group permission asset-dev read-only

#### Permission Verification Code

```python
import json
from system.models import Role
from functools import wraps
from django.shortcuts import HttpResponse


def role_permission_get_list(function):
    """
    Control permissions on the list page
    :param function:
    :return:
    """

    @wraps(function)
    def wrapped(self):
        user = self.request.user
        groups = [x['name'] for x in self.request.user.groups.values()]
        request_type = self.request.method
        model = str(self.model._meta).split(".")[1]

        filter_dict = {}
        not_list = ['page', 'order_by', 'csrfmiddlewaretoken']
        for k, v in dict(self.request.GET).items():
            if [i for i in v if i != ''] and (k not in not_list):
                if '__in' in k:
                    filter_dict[k] = v
                else:
                    filter_dict[k] = v[0]

        if not user.is_superuser:
            role_groups = Role.objects.filter(group__name__in=groups).values_list('roles__table',
                                                                                  'roles__request__request',
                                                                                  'roles__permission')

            permission_dict = {}
            for i in role_groups:
                if i[0] == model and i[1] == request_type:
                    permission_dict = json.loads(i[2])

            if permission_dict:
                if filter_dict:
                    for k, v in permission_dict.items():
                        if '__in' in k:
                            k1 = k.replace('__in', '')
                        if '__gt' in k:
                            k1 = k.replace('__gt', '')
                        if '__lt' in k:
                            k1 = k.replace('__lt', '')
                        else:
                            k1 = k
                        if k1 in list(filter_dict.keys()):
                            del filter_dict[k1]

                    if filter_dict:
                        filter_dict.update(**permission_dict)
                    else:
                        print('查询条件处理后为空,默认权限')
                        filter_dict = permission_dict
                else:
                    print('查询条件为空,默认权限')
                    filter_dict = permission_dict
            else:
                print('没有权限')
                filter_dict = {'id': -1}

        self.filter_dict = filter_dict
        result = function(self)
        return result

    return wrapped


def role_permission_detail(function):
    """
    Control permissions on the detail page
    :param function:
    :return:
    """

    @wraps(function)
    def wrapped(self, request, *args, **kwargs):
        user = self.request.user

        if not user.is_superuser:
            groups = [x['name'] for x in self.request.user.groups.values()]
            request_type = self.request.method
            model = str(self.model._meta).split(".")[1]
            pk = self.kwargs.get(self.pk_url_kwarg, None)

            role_groups = Role.objects.filter(group__name__in=groups).values_list('roles__table',
                                                                                  'roles__request__request',
                                                                                  'roles__permission')

            permission_dict = {}
            for i in role_groups:
                if i[0] == model and i[1] == request_type:
                    permission_dict = json.loads(i[2])

            permission_dict['id'] = pk
            obj = self.model.objects.filter(**permission_dict).count()
            if not obj:
                return HttpResponse(status=403)

        result = function(self, request, *args, **kwargs)
        return result

    return wrapped


def role_permission_update_delete(function):
    """
    Control permissions on the detail page
    :param function:
    :return:
    """

    @wraps(function)
    def wrapped(self, request):
        user = self.request.user
        if not user.is_superuser:

            groups = [x['name'] for x in self.request.user.groups.values()]
            request_type = self.request.method
            model = str(self.model._meta).split(".")[1]
            pk = self.request.POST.get('nid', None)

            role_groups = Role.objects.filter(group__name__in=groups).values_list('roles__table',
                                                                                  'roles__request__request',
                                                                                  'roles__permission')

            permission_dict = {}
            for i in role_groups:
                if i[0] == model and i[1] == request_type:
                    permission_dict = json.loads(i[2])

            permission_dict['id'] = pk
            obj = self.model.objects.filter(**permission_dict).count()
            if not obj:
                ret = {'status': None, 'error': "没有权限,拒绝", 'msg': 'Without permission, rejected'}
                return HttpResponse(json.dumps(ret))

        result = function(self, request)
        return result

    return wrapped
```

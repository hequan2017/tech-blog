f(a = 1, c = 3)                    # a mix of keyword and default arguments

def keyOnly(a, *b, c): print('')   # c is keyword-only; it must be matched as c = value
def keyOnly(a, *, b, c): ......    # b and c are keyword-only; they must be matched by keyword
def keyOnly(a, *, b = 1): ......   # b has a default: omit it, or pass it as the keyword argument b = value

def f(a, *b, **c): print(a, b, c)  # both combined
f(1, 2, 3, x=4, y=5)               # prints 1, (2, 3), {'x':4, 'y':5}

#-- Arbitrary-depth assignment in Python, e.g. a[0] = 'value1'; a[1][2] = 'value2'; a[3][4][5] = 'value3'
class MyDict(dict):
def __setitem__(self, key, value):                 # this function makes no changes; it is here just for the output
print('setitem:', key, value, self)
super().__setitem__(key, value)
def __getitem__(self, item):                       # the main trick is in this function
print('getitem:', item, self)                  # print information

# basic idea: when assigning a[1][2], a[1] must be fetched first, and then its [2] is assigned
if item not in self:                           # if a[1] does not exist, a new dict must be created and a[1] = dict set
temp = MyDict()                            # the newly created dict: temp
super().__setitem__(item, temp)            # assign a[1] = temp
return temp                                # return temp so that temp[2] = value works
return super().__getitem__(item)               # if a[1] exists, return a[1] directly

# example:
test = MyDict()
test[0] = 'test'
print(test[0])
test[1][2] = 'test1'
print(test[1][2])
test[1][3] = 'test2'

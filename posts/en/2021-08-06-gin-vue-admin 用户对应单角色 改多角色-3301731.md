> **About this post**
>
> On top of an older version of gin-vue-admin, this post changes "one user, one role" into multiple roles per user: a `many2many` association through the `sys_users_authority_id` join table, a role list carried in the JWT, menus merged and de-duplicated across multiple roles when queried, and the Casbin interceptor iterating over all roles for permission checks.

> **Technical notes**
>
> The official newer versions of gin-vue-admin already support multiple roles natively, so this post is mainly for reference if you are still maintaining an older version. The examples are based on GORM v1 and the old project structure; when upgrading, it is recommended to align directly with the latest official version instead of maintaining the join table and permission logic by hand.

#### The newer version already supports multiple roles.

```go
func GetMenu(c *gin.Context) {
	_, idList := getUserAuthorityId(c)
	if err, menus := service.GetMenuTree(idList); err != nil {
		global.GVA_LOG.Error("获取失败!", zap.Any("err", err))
		response.FailWithMessage("获取失败", c)
	} else {
		if menus == nil {
			menus = []model.SysMenu{}
		}
		var menuList []model.SysMenu
		temp := map[uint]struct{}{}
		for _, v := range menus {
			if _, ok := temp[]; !ok {
				temp[] = struct{}{}
				var menuListTemp []model.SysMenu
				temp1 := map[uint]struct{}{}
				for _, v1 := range v.Children {
					if _, ok := temp1[]; !ok {
						temp1[] = struct{}{}
						menuListTemp = append(menuListTemp, v1)
					}
				}
				v.Children = menuListTemp
				menuList = append(menuList, v)
			} else {
			}
		}
		response.OkWithDetailed(response.SysMenusResponse{Menus: menuList}, "获取成功", c)
	}
}
```

```go
// Get the user role IDs parsed from the JWT out of the Gin context
func getUserAuthorityId(c *gin.Context) (string, []string) {
	if claims, exists := c.Get("claims"); !exists {
		global.GVA_LOG.Error("从Gin的Context中获取从jwt解析出来的用户UUID失败, 请检查路由是否使用jwt中间件!")
		return "", []string{}
	} else {
		waitUse := claims.(*request.CustomClaims)
		return waitUse.AuthorityId, waitUse.AuthorityList
	}
}
```

#### Permission check

```go
// Interceptor
func CasbinHandler() gin.HandlerFunc {
	return func(c *gin.Context) {
		claims, _ := c.Get("claims")
		waitUse := claims.(*request.CustomClaims)
		// Get the request URI
		obj := c.Request.URL.RequestURI()
		// Get the request method
		act := c.Request.Method
		// Get the user's roles

		e := service.Casbin()
		// Check whether the policy allows it
		p := false
		for _, v := range waitUse.AuthorityList {
			success, _ := e.Enforce(v, obj, act)
			if global.GVA_CONFIG.System.Env == "develop" || success {
				p = true
			}
		}
		if p {
			c.Next()
		} else {
			response.FailWithDetailed(gin.H{}, "权限不足", c)
			c.Abort()
			return
		}
	}
}
```

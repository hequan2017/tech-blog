> **About this post**
>
> This post presents the complete code of the `ppp` package in the go-ppp
> project of the dial-up management system: it defines structs such as
> PppData (account, password, bandwidth, Vlanid, count, Mac, Nic) and the
> pagination query parameters, and implements CRUD operations for dial-up
> accounts on top of GORM: Create, Delete, batch delete by ids, Update,
> single lookup by id, and the GetPppInfoList paginated list, with fuzzy
> search by account keyword.

> **Technical notes**
>
> Write operations are completed in a single line via db.Create / Delete /
> Save, and batch deletion uses the "id in ?" condition; the list query
> starts the chain with Model, first calls Count to get the total, then
> pages with Limit/Offset and fetches the results with Find. Method
> signatures keep the old-style (err, data) return order, and the
> PageInfo / IdsReq / GetXxxInfoList structure matches the service
> templates auto-generated by gin-vue-admin. Note: the list passes the
> keyword via info.Account, while the matched column name in SQL is sn;
> fields such as Vlanid / Mac / Nic in PppData correspond to the inputs
> needed for creating the VLAN, macvlan, and ppp configuration in the
> "Create PPP Dial-up" script of this series.

## 1. Complete Code of the ppp Package

The whole file is the ppp package: it first defines the data structures and pagination parameters, then provides six CRUD methods:

```go
package ppp

import (
    "gorm.io/gorm"
)

type PppService struct {
}

type PppSearch struct {
    PppData
    PageInfo
}

type IdsReq struct {
    Ids []int `json:"ids" form:"ids"`
}

type PageInfo struct {
    Page     int    `json:"page" form:"page"`         // page number
    PageSize int    `json:"pageSize" form:"pageSize"` // page size
    Keyword  string `json:"keyword" form:"keyword"`   // keyword
}

type PppData struct {
    Account   string
    Password  string
    Bandwidth string
    Vlanid    int
    Count     int
    Mac       string
    Nic       string
}

var db *gorm.DB

func (PppService *PppService) CreatePpp(ppp PppData) (err error) {
    err = db.Create(&ppp).Error
    return err
}

func (PppService *PppService) DeletePpp(ppp PppData) (err error) {
    err = db.Delete(&ppp).Error
    return err
}

func (PppService *PppService) DeletePppByIds(ids IdsReq) (err error) {
    err = db.Delete(&[]PppData{}, "id in ?", ids.Ids).Error
    return err
}

func (PppService *PppService) UpdatePpp(ppp PppData) (err error) {
    err = db.Save(&ppp).Error
    return err
}

func (PppService *PppService) GetPpp(id uint) (err error, ppp PppData) {
    err = db.Where("id = ?", id).First(&ppp).Error
    return
}

func (PppService *PppService) GetPppInfoList(info PppSearch) (err error, list interface{}, total int64) {
    limit := info.PageSize
    offset := info.PageSize * (info.Page - 1)
    // create db
    db1 := db.Model(&PppData{})
    var ppps []PppData
    // if there is a search condition, the query is built automatically below
    if info.Account != "" {
        db1 = db1.Where("sn LIKE ?", "%"+info.Account+"%")
    }
    err = db1.Count(&total).Error
    if err != nil {
        return
    }
    err = db1.Limit(limit).Offset(offset).Find(&ppps).Error
    return err, ppps, total
}
```

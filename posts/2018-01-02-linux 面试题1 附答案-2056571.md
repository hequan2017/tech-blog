14、写一个脚本，实现批量添加20个用户，用户名为：user1-20，密码为 user 后面跟着5个随机字符或数字

for i in `seq 1 20`
do
pw=`echo $[$RANDOM]|md5sum|cut -c 1-5`
useradd user$i
echo "user$i $pw" >> /root/pw.txt
echo "user$pw" |passwd --stdin user$i
done

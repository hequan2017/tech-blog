> **内容介绍**
>
> 本文用 Go 演示基于 Redis Pub/Sub 的"生产者消费者"：
> 生产者协程循环向 queue:message 频道 Publish 1000 条消息；
> 消费者用 Subscribe 订阅该频道，从 Channel() 中持续接收并打印；
> 文末附完整示例代码，基于 github.com/go-redis/redis/v8。

> **技术备注**
>
> 示例虽名为生产者消费者，实际用的是 Pub/Sub：消息"即发即弃"，只投递给发布时在线的订阅者，不落盘、无确认与重试，订阅者不在线或中途掉线都会丢消息，并不适合做可靠队列。
> 可靠队列的惯用法是 List 的 LPUSH + BRPOPLPUSH（阻塞弹出并原子写入备份队列，宕机未处理完可从备份恢复），Redis 5.0+ 也可改用 Stream（XADD/XREADGROUP，支持消费组与 ACK）。
> 消费者外层的 for{} 用于订阅断开（Channel() 关闭）后重新 Subscribe，继续消费。
> go-redis 自 v8 起模块路径为 github.com/go-redis/redis/v8，且所有命令需显式传入 context；v9 已迁移至 github.com/redis/go-redis/v9。

## 1. 完整示例代码

```go
package main

import (
    "context"
    "github.com/go-redis/redis/v8"
    "log"
)

var ctx = context.Background()

func ProducerMessageList(rdb *redis.Client, queueListKey string) {
    for i := 0; i < 1000; i++ {
        _, err := rdb.Publish(ctx, queueListKey, i).Result()
        if err != nil {
            log.Println(err)
        }
    }
}

func ConsumerMessageList(rdb *redis.Client, queueListKey string) {
    for {
        pubSub := rdb.Subscribe(ctx, queueListKey)
        ch := pubSub.Channel()
        for msg := range ch {
            log.Printf("消费到数据 channel:%s；message:%s\n", msg.Channel, msg.Payload)
        }
    }
}

func main() {

    rdb := redis.NewClient(&redis.Options{
        Addr:     "192.168.117.129:6379",
        Password: "foobared", // no password set
        DB:       0,          // use default DB
    })
    // 测试连接是否成功
    _ = rdb.Ping(context.Background()).Err()

    go ProducerMessageList(rdb, "queue:message")

    ConsumerMessageList(rdb, "queue:message")

}
```

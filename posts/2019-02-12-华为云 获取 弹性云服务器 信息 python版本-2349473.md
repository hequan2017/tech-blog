#### 安装SDK

#### 代码

```pythonfrom openstack import connection

# create connection
username = "hequan"
password = "hequan"
projectId = "xxxxxxxxxxxxxxxxxxxxxxxxx"  # tenant ID
userDomainId = "xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx"  # user account ID
auth_url = "https://iam.cn-north-1.myhuaweicloud.com"  # endpoint url https://ecs.cn-north-1.myhuaweicloud.com
conn = connection.Connection(auth_url=auth_url,
                             user_domain_id=userDomainId,
                             project_id=projectId,
                             username=username,
                             password=password)

# get list of server
def list_servers():
    servers = conn.compute.servers(limit=1)
    for server in servers:
        print(server)

if __name__ == "__main__":
    list_servers()
```

> **技术备注**
>
> CentOS 7 已于 2024 年 6 月 30 日停止维护(EOL),建议迁移至 Rocky Linux 9 / AlmaLinux 9 或国产 openEuler。

```shell
egrep -o '(vmx|svm)' /proc/cpuinfo

sudo dnf install -y qemu-kvm libvirt libvirt-daemon virt-install virt-manager

sudo systemctl enable --now libvirtd
sudo systemctl start libvirtd

lsmod | grep kvm

qemu-img create -f raw cenos7.raw 10G
virt-install --name centos7 --vcpus 1 --ram 1024 --location=/vm-iso/CentOS-7-x86_64-Minimal-2009.iso --disk path=/vm-centos7/cenos7.raw,size=10,format=raw --network=bridge=virbr0 --os-type=linux --os-variant=rhel7 --extra-args='console=ttyS0' --noautoconsole --force

virsh list --all
virsh dominfo centos7
virsh console centos7

virsh start centos7
virsh shutdown centos7
virsh undefine centos7 --remove-all-storage

qemu-img create -f qcow2 -b cenos7.raw cenos7-02.qcow2         # 为cenos7-02虚拟机创建新的虚拟机硬盘镜像

cd /etc/libvirt/qemu
cp centos7.xml centos7-02.xml 
vi centos7-02.xml 

  <name>cenos7-02</name>
  <uuid>322f0f23-cdb5-4097-afc1-79636dd8fe2f</uuid>
      <source file='/vm-centos7/cenos7-02.qcow2
       <mac address='52:54:00:87:c4:b2

virsh define centos7-02.xml  注册

第二种:
virt-clone -o centos7 -n vm2 -f /var/lib/libvirt/images/vm2.qcow2

转化格式:   将raw格式转换为qcow2
qemu-img convert -f raw -O qcow2 /vm-cenos7/centos7.raw  /vm-cenos7centos7.qcow2        

导入：
sudo virt-install \
  --name vm3 \
  --ram 2048 \
  --vcpus 2 \
  --os-variant centos7.0 \
  --disk path=/var/lib/libvirt/images/vm3.qcow2,format=qcow2 \
  --network bridge=br1 \
  --graphics none \
  --console pty,target_type=serial \
  --import

网络桥接
cp ifcfg-ens160 ifcfg-br1

vim ifcfg-ens160
BRIDGE=br1

vim ifcfg-br1
TYPE=Bridge
PROXY_METHOD=none
BROWSER_ONLY=no
BOOTPROTO=dhcp
DEFROUTE=yes
IPV4_FAILURE_FATAL=no
ONBOOT=yes
NAME=br1
DEVICE=br1

 dnf install https://dl.fedoraproject.org/pub/epel/epel-release-latest-8.noarch.rpm
 dnf install bridge-utils

brctl show
virsh domiflist centos7

 brctl delif virbr0  vnet0
brctl addif br1  vnet0

brctl show
```

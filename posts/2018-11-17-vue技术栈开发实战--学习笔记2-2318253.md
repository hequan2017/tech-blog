#### 2 路由基础

- `<router-link>` 对应 `<a>`，`<router-view/>` 用于渲染组件。

路由对象常用属性：

```js
{
  path: '',
  component: {},        // 或 components
  children: [
    { /* 子集 */ }
  ],
  name: 'about',        // 命名，配合 :to="{ name: 'about' }"
  alias: '/别名',
  redirect: to => '/'   // 重定向
}
```

动态参数：`path: '/argu/:name'`，组件内通过 `$route.params.name` 获取。

编程式导航 `this.$router`：

```js
this.$router.back()          // 或 .go(-1)

  name: 'argu',
  params: {
    name: 'lison'
  }
})

  path: `/argu/${name}`
})

this.$router.replace({ /* 同上 */ })
```

#### 3 路由进阶

路由组件传参：

```js
// 布尔模式：把 params 作为 props 传给组件
props: true

// 对象模式
props: {
  food: 'banana'
}

// 函数模式：/?food=xxxx
props: route => ({
  food: route.query.food
})
```

完整的导航解析流程：

* 1. 导航被触发
* 2. 在失活的组件（即将离开的页面组件）里调用离开守卫 beforeRouteLeave
* 3. 调用全局的前置守卫 beforeEach
* 4. 在重用的组件里调用 beforeRouteUpdate
* 5. 调用路由独享的守卫 beforeEnter
* 6. 解析异步路由组件
* 7. 在被激活的组件（即将进入的页面组件）里调用 beforeRouteEnter
* 8. 调用全局的解析守卫 beforeResolve
* 9. 导航被确认
* 10. 调用全局的后置守卫 afterEach
* 11. 触发DOM更新
* 12. 用创建好的实例调用 beforeRouteEnter 守卫里传给 next 的回调函数

meta 元信息与全局守卫示例：

```js
// 路由配置
meta: {
    title: '元信息'
}

router.beforeEach((to, from, next) => {
    to.meta && setTitle(to.meta.title)
})
// to   对象 即将跳转
// from 对象 离开的页面
// next 函数 跳转
```

#### 6 vuex-mutation_action_module

```js
import { mapMutations, mapActions } from 'vuex'

const mutations = {
  SET_APP_NAME (state, params) {
    state.appName = params
  },
}

// action
async updateAppName ({ commit }) {
  try {
    const { info: { appName } } = await getAppName()
    commit('SET_APP_NAME', appName)
  } catch (err) {
    console.log(err)
  }
}

methods: {
  ...mapMutations([  // 同步
    'SET_APP_NAME'
  ]),
  ...mapActions([    // 异步
    'updateAppName'
  ]),
  handleChangeAppName () {
    this.$store.commit({
      type: 'SET_APP_NAME',
      appName: 'newAppName'
    })
    this.SET_APP_NAME({
      appName: 'newAppName'
    })
    this.$store.dispatch('updateAppName', '123')
  }
}

// 如果 state 初始化没有该变量，需要用 set，才会增加响应式的 set/get
vue.set(state, 'appVersion', 'v2.0')
```

user 模块：登录后获取权限并保存组件级权限：

```js
import { login, authorization } from '@/api/user'
import { setToken } from '@/lib/util'

const state = {
  userName: 'Lison',
  rules: {}   // 组件级权限 存储
}

const mutations = {
  SET_RULES (state, rules) {
    state.rules = rules
  }
}

const actions = {
  authorization ({ commit }, token) {
    return new Promise((resolve, reject) => {
      authorization().then(res => {
        if (parseInt(res.code) === 401) {
          reject(new Error('token error'))
        } else {
          setToken(res.data.token)
          resolve(res.data.rules.page)
          commit('SET_RULES', res.data.rules.component)
        }
      }).catch(error => {
        reject(error)
      })
    })
  }
}
```

router 模块：按权限过滤并合并路由：

```js
import { routes, routerMap } from '@/router/router'

const state = {
  routers: routes,
  hasGetRules: false
}

const mutations = {
  CONCAT_ROUTES (state, routerList) {
    state.routers = routerList.concat(routes)    // 合并路由
    state.hasGetRules = true   // 已经获取到 路由权限
  }
}

const getAccesRouterList = (routes, rules) => {
  return routes.filter(item => {
    if (rules[item.name]) {
      if (item.children) item.children = getAccesRouterList(item.children, rules)
      return true
    } else return false
  })
}

const actions = {
  concatRoutes ({ commit }, rules) {
    return new Promise((resolve, reject) => {
      try {
        let routerList = []
        if (Object.entries(rules).every(item => item[1])) {
          routerList = routerMap
        } else {
          routerList = getAccesRouterList(routerMap, rules)  // 可以访问的路由列表
        }
        commit('CONCAT_ROUTES', routerList)   // 设置路由
        resolve(routerList)
      } catch (err) {
        reject(err)
      }
    })
  }
}
```

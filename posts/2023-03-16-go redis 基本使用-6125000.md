package main

import (
    "context"
    "/go-redis/redis/v8"
    "log"
    "math/rand"
    "time"
)

var ctx = context.Background()

func ProducerMessageList(rdb *redis.Client, queueListKey string) {
    rand.Seed(time.Now().UnixNano())
    log.Println("开启生产者。。。。")

    for i := 0; i < 10000; i++ {
        score := time.Now().Unix()
        log.Println("正在生产一条消息...", score, i)
        _, err := rdb.LPush(ctx, queueListKey, i).Result()
        if err != nil {
            log.Println(err)
        }
        //time.Sleep(time.Duration(rand.Intn(3)) * time.Second)
    }
}

func ConsumerMessageList(rdb *redis.Client, queueListKey string) {
    for {
        // 设置一个5秒的超时时间
        value, err := rdb.BRPop(ctx, 5*time.Second, queueListKey).Result()
        if err == redis.Nil {
            time.Sleep(1 * time.Second)
            continue
        }
        if err != nil {
            time.Sleep(1 * time.Second)
            continue
        }
        log.Println("消费到数据：", value, "当前时间是：", time.Now().Unix())
        //time.Sleep(time.Second)
    }
}

func main() {

    rdb := redis.NewClient(&redis.Options{
        Addr:     "192.168.117.129:6379",
        Password: "foobared", // no password set
        DB:       0,          // use default DB
    })
    // 测试连接是否成功
    _ = rdb.Ping(context.Background()).Err()

    //_ = rdb.Set(ctx, "key", "123", 0).Err()
    //
    //val, _ := rdb.Get(ctx, "key").Result()
    //
    //fmt.Println("key", val)

    go ProducerMessageList(rdb, "queue:message")
    ConsumerMessageList(rdb, "queue:message")

}

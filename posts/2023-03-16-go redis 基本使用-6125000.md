> **内容介绍**
>
> 本文用 Go 操作 Redis，实现一个最简版的消息队列：
> 生产者用 LPush 向 queue:message 列表写入 1 万条消息；
> 消费者用 BRPop 阻塞式弹出，超时设为 5 秒，并对 redis.Nil 做专门处理；
> 文末附完整示例代码，基于 github.com/go-redis/redis/v8。

> **技术备注**
>
> go-redis 自 v8 起模块路径变为 github.com/go-redis/redis/v8，且所有命令都要求显式传入 context.Context（文中统一用 context.Background()）；v9 已迁移至 github.com/redis/go-redis/v9。
> BRPop 在超时且未取到数据时返回 redis.Nil，这是正常情况，应与真实错误分开处理。
> 文中 rand.Seed 自 Go 1.20 起已废弃，现代码无需手动播种。

## 1. 完整示例代码

```go
package main

import (
    "context"
    "github.com/go-redis/redis/v8"
    "log"
    "math/rand"
    "time"
)

var ctx = context.Background()

func ProducerMessageList(rdb *redis.Client, queueListKey string) {
    rand.Seed(time.Now().UnixNano())
    log.Println("开启生产者。。。。")

    for i := 0; i < 10000; i++ {
        score := time.Now().Unix()
        log.Println("正在生产一条消息...", score, i)
        _, err := rdb.LPush(ctx, queueListKey, i).Result()
        if err != nil {
            log.Println(err)
        }
        //time.Sleep(time.Duration(rand.Intn(3)) * time.Second)
    }
}

func ConsumerMessageList(rdb *redis.Client, queueListKey string) {
    for {
        // 设置一个5秒的超时时间
        value, err := rdb.BRPop(ctx, 5*time.Second, queueListKey).Result()
        if err == redis.Nil {
            time.Sleep(1 * time.Second)
            continue
        }
        if err != nil {
            time.Sleep(1 * time.Second)
            continue
        }
        log.Println("消费到数据：", value, "当前时间是：", time.Now().Unix())
        //time.Sleep(time.Second)
    }
}

func main() {

    rdb := redis.NewClient(&redis.Options{
        Addr:     "192.168.117.129:6379",
        Password: "foobared", // no password set
        DB:       0,          // use default DB
    })
    // 测试连接是否成功
    _ = rdb.Ping(context.Background()).Err()

    //_ = rdb.Set(ctx, "key", "123", 0).Err()
    //
    //val, _ := rdb.Get(ctx, "key").Result()
    //
    //fmt.Println("key", val)

    go ProducerMessageList(rdb, "queue:message")
    ConsumerMessageList(rdb, "queue:message")

}
```

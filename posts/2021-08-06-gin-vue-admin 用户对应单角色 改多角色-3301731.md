GetMenu(c *gin.Context) {
	_, idList := getUserAuthorityId(c)
	if err, menus := service.GetMenuTree(idList); err != nil {
		global.GVA_LOG.Error("获取失败!", zap.Any("err", err))
		response.FailWithMessage("获取失败", c)
	} else {
		if menus == nil {
			menus = []model.SysMenu{}
		}
		var menuList []model.SysMenu
		temp := map[uint]struct{}{}
		for _, v := range menus {
			if _, ok := temp[]; !ok {
				temp[] = struct{}{}
				var menuListTemp []model.SysMenu
				temp1 := map[uint]struct{}{}
				for _, v1 := range v.Children {
					if _, ok := temp1[]; !ok {
						temp1[] = struct{}{}
						menuListTemp = append(menuListTemp, v1)
					}
				}
				v.Children = menuListTemp
				menuList = append(menuList, v)
			} else {
			}
		}
		response.OkWithDetailed(response.SysMenusResponse{Menus: menuList}, "获取成功", c)
	}
}

```go
// 从Gin的Context中获取从jwt解析出来的用户角色id
func getUserAuthorityId(c *gin.Context) (string, []string) {
	if claims, exists := c.Get("claims"); !exists {
		global.GVA_LOG.Error("从Gin的Context中获取从jwt解析出来的用户UUID失败, 请检查路由是否使用jwt中间件!")
		return "", []string{}
	} else {
		waitUse := claims.(*request.CustomClaims)
		return waitUse.AuthorityId, waitUse.AuthorityList
	}
}
```

#### 权限判断

```go
// 拦截器
func CasbinHandler() gin.HandlerFunc {
	return func(c *gin.Context) {
		claims, _ := c.Get("claims")
		waitUse := claims.(*request.CustomClaims)
		// 获取请求的URI
		obj := c.Request.URL.RequestURI()
		// 获取请求方法
		act := c.Request.Method
		// 获取用户的角色

		e := service.Casbin()
		// 判断策略中是否存在
		p := false
		for _, v := range waitUse.AuthorityList {
			success, _ := e.Enforce(v, obj, act)
			if global.GVA_CONFIG.System.Env == "develop" || success {
				p = true
			}
		}
		if p {
			c.Next()
		} else {
			response.FailWithDetailed(gin.H{}, "权限不足", c)
			c.Abort()
			return
		}
	}
}
```

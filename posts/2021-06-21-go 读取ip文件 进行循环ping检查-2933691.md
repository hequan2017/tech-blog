> **内容介绍**
>
> 用 Go 读取 `ip` 文件逐行解析地址，借助 `go-fastping` 库发送 ICMP Ping 探测，根据是否收到回包输出每台主机的 `ok/no` 状态，适合做简单的存活巡检脚本。

> **技术备注**
>
> 示例基于 `tatsushid/go-fastping` 编写；原文 import 路径被截断，实际应为 `github.com/tatsushid/go-fastping`。该库已多年未更新，Go 1.20+ 环境下可考虑迁移到 `github.com/prometheus-community/pro-bing` 或系统 `ping` 命令封装。

---

```go
package main

import (
	"fmt"
	"io/ioutil"
	"net"
	"os"
	"strings"
	"time"

	"github.com/tatsushid/go-fastping"
)

func main() {
	var result []string
	b, err := ioutil.ReadFile("ip")
	if err != nil {
	}
	s := string(b)
	for _, lineStr := range strings.Split(s, "\n") {
		lineStr = strings.TrimSpace(lineStr)
		if lineStr == "" {
			continue
		}
		result = append(result, lineStr)
	}
	for _, v := range result {
		p := fastping.NewPinger()
		ra, err := net.ResolveIPAddr("ip4:icmp", v)
		if err != nil {
			fmt.Println(err)
			os.Exit(1)
		}
		p.AddIPAddr(ra)
		var d int
		p.OnRecv = func(addr *net.IPAddr, rtt time.Duration) {
			d = 1
		}
		p.OnIdle = func() {
			if d == 1 {
				fmt.Println(v, "ok")
			} else {
				fmt.Println(v, "no")
			}
		}
		err = p.Run()
		if err != nil {
			fmt.Println(err)
		}
	}
}
```
